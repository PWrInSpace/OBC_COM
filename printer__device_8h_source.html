49// Clear the received FIFO
50void tud_printer_n_read_flush(uint8_t itf);

101// Application Callback API (weak is optional)
102//--------------------------------------------------------------------+
103
104// Invoked when received new data
105void tud_printer_rx_cb(uint8_t itf);
106
107// Invoked when last write transfer is completed
108void tud_printer_tx_complete_cb(uint8_t itf);
109
110// Invoked when host requests device ID string (IEEE 1284).
111// Application returns pointer to device ID buffer (must remain valid until transfer completes).
112// First 2 bytes of returned buffer must contain big-endian length (including the 2 length bytes).
113const uint8_t *tud_printer_get_device_id_cb(uint8_t itf);
114
115// Invoked when host requests port status.
