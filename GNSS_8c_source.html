52void GNSS_ParsePVTData(GNSS_StateHandle *GNSS) {
53 // Offsety z dokumentacji + 6 bajtów nagłówka UBX
54
55 // 1. Czas i Data
56 // Rok (offset 4 w Payloadzie) -> indeks 10
57 GNSS->year = (uint16_t)(GNSS->uartWorkingBuffer[10] |
58 (GNSS->uartWorkingBuffer[11] << 8));
59 GNSS->month = GNSS->uartWorkingBuffer[12]; // offset 6 -> indeks 12
60 GNSS->day = GNSS->uartWorkingBuffer[13]; // offset 7 -> indeks 13
61 GNSS->hour = GNSS->uartWorkingBuffer[14]; // offset 8 -> indeks 14
62 GNSS->min = GNSS->uartWorkingBuffer[15]; // offset 9 -> indeks 15
63 GNSS->sec = GNSS->uartWorkingBuffer[16]; // offset 10 -> indeks 16
64
65 // 2. Status Fixa i Satelity
66 // fixType (offset 20) -> indeks 26
67 GNSS->fixType = GNSS->uartWorkingBuffer[26];
68 // numSV (offset 23) -> indeks 29
69 GNSS->numSV = GNSS->uartWorkingBuffer[29];
70
71 // 3. Pozycja (Lon/Lat: 1e-7 deg)
72 // lon (offset 24) -> indeks 30
73 int32_t tempLon = (int32_t)(GNSS->uartWorkingBuffer[30] |
74 (GNSS->uartWorkingBuffer[31] << 8) |
75 (GNSS->uartWorkingBuffer[32] << 16) |
76 (GNSS->uartWorkingBuffer[33] << 24));
77 // lat (offset 28) -> indeks 34
78 int32_t tempLat = (int32_t)(GNSS->uartWorkingBuffer[34] |
79 (GNSS->uartWorkingBuffer[35] << 8) |
80 (GNSS->uartWorkingBuffer[36] << 16) |
81 (GNSS->uartWorkingBuffer[37] << 24));
82
83 GNSS->lon = tempLon;
84 GNSS->lat = tempLat;
85 GNSS->fLon = (float)tempLon / 10000000.0f;
86 GNSS->fLat = (float)tempLat / 10000000.0f;
87
88 // 4. Wysokość (hMSL: mm)
89 // hMSL (offset 36) -> indeks 42
90 GNSS->hMSL = (int32_t)(GNSS->uartWorkingBuffer[42] |
91 (GNSS->uartWorkingBuffer[43] << 8) |
92 (GNSS->uartWorkingBuffer[44] << 16) |
93 (GNSS->uartWorkingBuffer[45] << 24));
94
95 // 5. Prędkość Ground Speed (offset 60) -> indeks 66
96 GNSS->gSpeed = (int32_t)(GNSS->uartWorkingBuffer[66] |
97 (GNSS->uartWorkingBuffer[67] << 8) |
98 (GNSS->uartWorkingBuffer[68] << 16) |
99 (GNSS->uartWorkingBuffer[69] << 24));
100 GNSS->fGSpeedKmH = (float)GNSS->gSpeed * 0.0036f; // Konwersja mm/s na km/h
101
102 // 6. Kurs Heading of Motion (offset 64) -> indeks 70
103 int32_t tempHead = (int32_t)(GNSS->uartWorkingBuffer[70] |
104 (GNSS->uartWorkingBuffer[71] << 8) |
105 (GNSS->uartWorkingBuffer[72] << 16) |
106 (GNSS->uartWorkingBuffer[73] << 24));
107 GNSS->headMot = (float)tempHead / 100000.0f;
108}

118void GNSS_CalcChecksum(uint8_t *data, uint8_t len, uint8_t *ck_a,
119 uint8_t *ck_b) {
120 *ck_a = 0;
121 *ck_b = 0;
122 for (uint8_t i = 2; i < len - 2;
123 i++) { // Pomijamy 0xB5 0x62 i ostatnie 2 bajty sumy
124 *ck_a = *ck_a + data[i];
125 *ck_b = *ck_b + *ck_a;
126 }
127}
