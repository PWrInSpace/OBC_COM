764// return payload buffer for next write
765TU_ATTR_ALWAYS_INLINE static inline uint8_t* mtp_container_payload_ptr(mtp_container_info_t* p_container) {
766 // only 1st packet include header
767 uint32_t pos = p_container->header->len - sizeof(mtp_container_header_t);
768 while (pos > CFG_TUD_MTP_EP_BUFSIZE) {
769 pos -= CFG_TUD_MTP_EP_BUFSIZE;
770 }
771 return p_container->payload + pos;
772}
773
774// only add_raw does partial copy
775TU_ATTR_ALWAYS_INLINE static inline uint32_t mtp_container_add_raw(mtp_container_info_t* p_container, const void* data, uint32_t len) {
776 uint8_t* buf = mtp_container_payload_ptr(p_container);
777 const uint32_t added_len = tu_min32(len, CFG_TUD_MTP_EP_BUFSIZE - p_container->header->len);
778 if (added_len > 0) {
779 memcpy(buf, data, added_len);
780 }
781 p_container->header->len += len; // always increase len, even partial copy
782 return added_len;
783}
784
785TU_ATTR_ALWAYS_INLINE static inline uint32_t mtp_container_add_array(mtp_container_info_t* p_container, uint8_t scalar_size, uint32_t count, const void* data) {
786 const uint32_t added_len = 4 + count * scalar_size;
787 TU_ASSERT(p_container->header->len + added_len < CFG_TUD_MTP_EP_BUFSIZE, 0);
788 uint8_t* buf = p_container->payload + p_container->header->len - sizeof(mtp_container_header_t);
789
790 tu_unaligned_write32(buf, count);
791 p_container->header->len += 4;
792 buf += 4;
793
794 memcpy(buf, data, count * scalar_size);
795 p_container->header->len += count * scalar_size;
796
797 return added_len;
798}
799
800TU_ATTR_ALWAYS_INLINE static inline uint32_t mtp_container_add_string(mtp_container_info_t* p_container, uint16_t* utf16) {
801 uint8_t count = 0;
802 while (utf16[count] != 0u) {
803 count++;
804 }
805 const uint32_t added_len = 1u + (uint32_t) count * 2u;
806 TU_ASSERT(p_container->header->len + added_len < CFG_TUD_MTP_EP_BUFSIZE, 0);
807 uint8_t* buf = p_container->payload + p_container->header->len - sizeof(mtp_container_header_t);
808
809 *buf++ = count;
810 p_container->header->len++;
811
812 memcpy(buf, utf16, 2u * (uint32_t) count);
813 p_container->header->len += 2u * count;
814
815 return added_len;
816}
817
818TU_ATTR_ALWAYS_INLINE static inline uint32_t mtp_container_add_cstring(mtp_container_info_t* p_container, const char* str) {
819 const uint8_t len = (uint8_t) (strlen(str) + 1); // include null
820 TU_ASSERT(p_container->header->len + 1 + 2 * len < CFG_TUD_MTP_EP_BUFSIZE, 0);
821 uint8_t* buf = p_container->payload + p_container->header->len - sizeof(mtp_container_header_t);
822
823 if (len == 1) {
824 // empty string (null only): single zero byte
825 *buf = 0;
826 p_container->header->len++;
827 return 1u;
828 } else {
829 *buf++ = len;
830 p_container->header->len++;
831
832 for (uint8_t i = 0; i < len; i++) {
833 buf[0] = str[i];
834 buf[1] = 0;
835 buf += 2;
836 p_container->header->len += 2;
837 }
838 return 1u + 2u * len;
839 }
840}
