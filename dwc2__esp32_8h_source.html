85static void dwc2_int_handler_wrap(void* arg) {
86 const uint8_t rhport = tu_u16_low((uint16_t)(uintptr_t)arg);
87 const tusb_role_t role = (tusb_role_t) tu_u16_high((uint16_t)(uintptr_t)arg);
88#if CFG_TUD_ENABLED
89 if (role == TUSB_ROLE_DEVICE) {
90 dcd_int_handler(rhport);
91 }
92#endif
93#if CFG_TUH_ENABLED && !CFG_TUH_MAX3421
94 if (role == TUSB_ROLE_HOST) {
95 hcd_int_handler(rhport, true);
96 }
97#endif
98}

100TU_ATTR_ALWAYS_INLINE static inline void dwc2_int_set(uint8_t rhport, tusb_role_t role, bool enabled) {
101 if (enabled) {
102 esp_intr_alloc(_dwc2_controller[rhport].irqnum, ESP_INTR_FLAG_LOWMED,
103 dwc2_int_handler_wrap, (void*)(uintptr_t)tu_u16(role, rhport), &usb_ih[rhport]);
104 } else {
105 esp_intr_free(usb_ih[rhport]);
106 }
107}

150TU_ATTR_ALWAYS_INLINE static inline uint32_t round_up_to_cache_line_size(uint32_t size) {
151 if (size & (CONFIG_CACHE_L1_CACHE_LINE_SIZE-1)) {
152 size = (size & ~(CONFIG_CACHE_L1_CACHE_LINE_SIZE-1)) + CONFIG_CACHE_L1_CACHE_LINE_SIZE;
153 }
154 return size;
155}

159 data_size = round_up_to_cache_line_size(data_size);
