112//--------------------------------------------------------------------+
113// Packet API
114//--------------------------------------------------------------------+
115
116// Read all available MIDI packets from the connected device
117// Return number of bytes read (always multiple of 4)
118uint32_t tuh_midi_packet_read_n(uint8_t idx, uint8_t *buffer, uint32_t bufsize);
119
120// Read a raw MIDI packet from the connected device
121// Return true if a packet was returned
122TU_ATTR_ALWAYS_INLINE static inline bool tuh_midi_packet_read(uint8_t idx, uint8_t packet[4]) {
123 return 4 == tuh_midi_packet_read_n(idx, packet, 4);
124}
125
126// Write all 4-byte packets, data is locally buffered and only transferred when buffered bytes
127// reach the endpoint packet size or tuh_midi_write_flush() is called
128uint32_t tuh_midi_packet_write_n(uint8_t idx, const uint8_t *buffer, uint32_t bufsize);
129
130// Write a 4-bytes packet to the device.
131// Returns true if the packet was successfully queued.
132TU_ATTR_ALWAYS_INLINE static inline bool tuh_midi_packet_write(uint8_t idx, const uint8_t packet[4]) {
133 return 4 == tuh_midi_packet_write_n(idx, packet, 4);
134}

146// Get the MIDI stream from the device. Set the value pointed
147// to by p_cable_num to the MIDI cable number intended to receive it.
148// The MIDI stream will be stored in the buffer pointed to by p_buffer.
149// Return the number of bytes added to the buffer.
150// Note that this function ignores the CIN field of the MIDI packet
151// because a number of commercial devices out there do not encode
152// it properly.
153uint32_t tuh_midi_stream_read(uint8_t idx, uint8_t *p_cable_num, uint8_t *p_buffer, uint16_t bufsize);
