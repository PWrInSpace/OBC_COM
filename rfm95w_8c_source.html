4rfm95_err_t rfm95_init(rfm95_t *rfm95) {
5 rfm95_err_t ret = RFM95_OK;
6
7 /*
8 * Perform hardware reset.
9 */
10 rfm95_reset(rfm95);
11
12 /*
13 * Check version.
14 */
15 uint8_t version;
16 uint8_t i = 0;
17 while (i++ < TIMEOUT_RESET) {
18 version = rfm95_read_reg(rfm95, REG_VERSION);
19 if (version == 0x12)
20 break;
21 rfm95->_delay(2);
22 }
23 assert(i <= TIMEOUT_RESET + 1); // at the end of the loop above, the max
24 // value i can reach is TIMEOUT_RESET + 1
25
26 ret |= rfm95_default_config(rfm95);
27
28 return ret;
29}

34rfm95_err_t rfm95_default_config(rfm95_t *rfm95) {
35 rfm95_err_t ret = RFM95_OK;
36 rfm95_sleep(rfm95);
37
38 // 1. Adresy bazowe FIFO
39 ret |= rfm95_write_reg(rfm95, REG_FIFO_RX_BASE_ADDR, 0);
40 ret |= rfm95_write_reg(rfm95, REG_FIFO_TX_BASE_ADDR, 0);
41
42 // 2. LNA i Modem Config
43 ret |= rfm95_write_reg(rfm95, REG_LNA, rfm95_read_reg(rfm95, REG_LNA) | 0x03);
44 ret |= rfm95_write_reg(rfm95, REG_MODEM_CONFIG_3, 0x04);
45
46 // 3. --- KONFIGURACJA PRZERWAŃ (DIO0) ---
47 // Mapujemy DIO0 na RX_DONE w trybie LoRa (Bity 7-6 na 00)
48 // Odczytujemy, zerujemy bity 7-6 i zapisujemy
49 uint8_t mapping = rfm95_read_reg(rfm95, REG_DIO_MAPPING_1);
50 mapping &= ~(0xC0); // Zerowanie bitów 7 i 6 -> DIO0 = RxDone
51 ret |= rfm95_write_reg(rfm95, REG_DIO_MAPPING_1, mapping);
52
53 // 4. --- ODBLOKOWANIE MASKI IRQ ---
54 // W REG_IRQ_FLAGS_MASK bit ustawiony na 0 oznacza WŁĄCZONE przerwanie.
55 // Chcemy RX_DONE (bit 6) i RX_TIMEOUT (bit 7) oraz CRC_ERR (bit 5).
56 // Zapisujemy 0x00, aby odblokować WSZYSTKIE przerwania LoRa.
57 ret |= rfm95_write_reg(rfm95, REG_IRQ_FLAGS_MASK, 0x00);
58
59 rfm95_set_tx_power(rfm95, RFM95_TX_POWER_17_dBm);
60
61 rfm95_idle(rfm95);
62 return ret;
63}

65rfm95_err_t rfm95_default_config_param(rfm95_t *rfm) {
66 rfm95_err_t ret = RFM95_OK;
67
68 // A. Musimy być w trybie Sleep, aby przełączyć się na LoRa
69 rfm95_sleep(rfm);
70
71 // B. Włączenie trybu LoRa (Bit 7 w REG_OP_MODE)
72 // Bez tego rejestry modemu nie będą dostępne
73 uint8_t opMode = rfm95_read_reg(rfm, 0x01);
74 ret |= rfm95_write_reg(rfm, 0x01, opMode | 0x80);
75
76 // C. Ustawienie częstotliwości (korzysta z rfm->param->frequency)
77 // Wymaga funkcji przeliczającej Hz na rejestry FrfMsb/Mid/Lsb
78 rfm95_set_frequency(rfm, rfm->param->frequency);
79 char buf[128] = {0};
80 sprintf(buf, "| Frequency | %-18lu Hz |\r\n",
81 (unsigned long)rfm->param->frequency);
82 USB_Transmit((uint8_t *)buf, strlen(buf));
83
84 // D. Konfiguracja FIFO
85 ret |= rfm95_write_reg(rfm, 0x0E, 0); // FIFO RX Base Addr
86 ret |= rfm95_write_reg(rfm, 0x0F, 0); // FIFO TX Base Addr
87
88 // E. Konfiguracja parametrów LoRa (BW i SF)
89 // REG_MODEM_CONFIG_1 (0x1D) -> Bandwidth
90 uint8_t bw_config = (rfm->param->LoRa_BW << 4) | 0x02; // + CR 4/5
91 ret |= rfm95_write_reg(rfm, 0x1D, bw_config);
92
93 // REG_MODEM_CONFIG_2 (0x1E) -> Spreading Factor (Rate)
94 uint8_t sf_config = (rfm->param->LoRa_Rate << 4);
95 ret |= rfm95_write_reg(rfm, 0x1E, sf_config);
96
97 // F. Ustawienie mocy nadawania
98 rfm95_set_tx_power(rfm, rfm->param->power);
99
100 if (rfm->param->crc) {
101 rfm95_enable_crc(rfm);
102 } else {
103 rfm95_disable_crc(rfm);
104 }
105
106 rfm95_err_t err = RFM95_OK;
107 err |= rfm95_set_coding_rate(rfm, rfm->param->CR);
108 err |= rfm95_set_sync_word(rfm, rfm->param->sync);
109 if (err != RFM95_OK) {
110 LOG_ERROR("Error setting CR or sync word in rfm95_default_config_param");
111 }
112
113 // G. Aktywacja automatycznego wzmocnienia (AGC)
114 ret |= rfm95_write_reg(rfm, 0x26, 0x04);
115
116 // H. Powrót do trybu Standby (aktywacja oscylatora kwarcowego)
117 rfm95_idle(rfm);
118
119 return ret;
120}

123 uint8_t out[2] = {0x80 | reg, val};
124 uint8_t in[2];

222rfm95_err_t rfm95_set_spreading_factor(rfm95_t *rfm95,
223 rfm95_spreading_factor_t sf) {
224 rfm95_err_t ret = RFM95_OK;
225 if (sf == RFM95_SF_64_CoS) {
226 ret |= rfm95_write_reg(rfm95, REG_DETECTION_THRESHOLD, 0x0c);
227 ret |= rfm95_write_reg(rfm95, REG_DETECTION_OPTIMIZE, 0xc5);
228 } else if (sf == RFM95_SF_4096_CoS) {
229 ret |= rfm95_write_reg(rfm95, REG_DETECTION_OPTIMIZE, 0xc3);
230 ret |= rfm95_write_reg(rfm95, REG_DETECTION_THRESHOLD, 0x0a);
231 }
232
233 ret |= rfm95_write_reg(rfm95, REG_MODEM_CONFIG_2,
234 (rfm95_read_reg(rfm95, REG_MODEM_CONFIG_2) & 0x0f) |
235 ((sf << 4) & 0xf0));
236 return ret;
237}

239rfm95_err_t rfm95_set_bandwidth(rfm95_t *rfm95, rfm95_bandwith_t sbw) {
240 if (sbw < 0 || sbw > 9) {
241 rfm95->log("ERROR: Setting bandwith unsuccessful: sbw out of range");
242 return RFM95_CONFIG_ERR;
243 }
244
245 if (sbw >= 8) {
246 int32_t freq = rfm95_get_frequency(rfm95);
247 if (freq <= 169E6) {
248 rfm95->log("INFO: In the set frequency set bandwith is not supported!");
249 return RFM95_CONFIG_ERR;
250 }
251 }
252
253 int16_t bw;
254 rfm95_err_t ret = RFM95_OK;
255 bw = (int16_t)sbw;
256
257 ret |= rfm95_write_reg(rfm95, REG_MODEM_CONFIG_1,
258 (rfm95_read_reg(rfm95, REG_MODEM_CONFIG_1) & 0x0f) |
259 (bw << 4));
260 return ret;
261}

312 int16_t size) {

342rfm95_err_t rfm95_send_packet(rfm95_t *rfm95, uint8_t *buf, int16_t size) {
343
344 rfm95_err_t ret = RFM95_OK;
345 ret |= rfm95_fill_fifo_buf_to_send(rfm95, buf, size);
346 ret |= rfm95_start_transmission(rfm95);
347
348 while (!rfm95_check_tx_done(rfm95)) {
349 int8_t read_reg = rfm95_read_reg(rfm95, REG_IRQ_FLAGS);
350 rfm95->_delay(2);
351 }
352
353 ret |= rfm95_write_irq_flags(rfm95);
354 return ret == RFM95_OK ? RFM95_OK : RFM95_TRANSMIT_ERR;
355}

357int16_t rfm95_receive_packet(rfm95_t *rfm95, uint8_t *buf, int16_t size) {
358 int16_t len = 0;
359
360 /*
361 * Check interrupts.
362 */
363 int16_t irq = rfm95_read_reg(rfm95, REG_IRQ_FLAGS);
364 rfm95_write_reg(rfm95, REG_IRQ_FLAGS, irq);
365 if ((irq & IRQ_RX_DONE_MASK) == 0)
366 return 0;
367 if (irq & IRQ_PAYLOAD_CRC_ERROR_MASK)
368 return 0;
369
370 /*
371 * Find packet size.
372 */
373 if (rfm95->implicit_header) {
374 len = rfm95_read_reg(rfm95, REG_PAYLOAD_LENGTH);
375 } else {
376 len = rfm95_read_reg(rfm95, REG_RX_NB_BYTES);
377 }
378
379 /*
380 * Transfer data from radio.
381 */
382 rfm95_idle(rfm95);
383 rfm95_write_reg(rfm95, REG_FIFO_ADDR_PTR,
384 rfm95_read_reg(rfm95, REG_FIFO_RX_CURRENT_ADDR));
385 if (len > size) {
386 len = size;
387 }
388 for (int16_t i = 0; i < len; i++) {
389 buf[i] = rfm95_read_reg(rfm95, REG_FIFO);
390 }
391
392 return len;
393}
