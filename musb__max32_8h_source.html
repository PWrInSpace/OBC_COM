68TU_ATTR_ALWAYS_INLINE static inline unsigned musb_dcd_get_int_enable(uint8_t rhport) {
69 #ifdef NVIC_GetEnableIRQ // only defined in CMSIS 5
70 return NVIC_GetEnableIRQ(musb_irqs[rhport]);
71 #else
72 uint32_t IRQn = (uint32_t) musb_irqs[rhport];
73 return ((NVIC->ISER[IRQn >> 5UL] & (1UL << (IRQn & 0x1FUL))) != 0UL) ? 1UL : 0UL;
74 #endif
75}

81static inline void musb_dcd_int_handler_enter(uint8_t rhport) {
82 mxc_usbhs_regs_t* hs_phy = MXC_USBHS;
83 uint32_t mxm_int, mxm_int_en, mxm_is;
84
85 //Handle PHY specific events
86 mxm_int = hs_phy->mxm_int;
87 mxm_int_en = hs_phy->mxm_int_en;
88 mxm_is = mxm_int & mxm_int_en;
89 hs_phy->mxm_int = mxm_is;
90
91 if (mxm_is & MXC_F_USBHS_MXM_INT_NOVBUS) {
92 dcd_event_bus_signal(rhport, DCD_EVENT_UNPLUGGED, true);
93 }
94}

96static inline void musb_dcd_phy_init(uint8_t rhport) {
97 (void) rhport;
98 mxc_usbhs_regs_t* hs_phy = MXC_USBHS;
99
100 // Interrupt for VBUS disconnect
101 hs_phy->mxm_int_en |= MXC_F_USBHS_MXM_INT_EN_NOVBUS;
102
103 musb_dcd_int_clear(rhport);
104
105 // Unsuspend the MAC
106 hs_phy->mxm_suspend = 0;
107
108 // Configure PHY
109 hs_phy->m31_phy_xcfgi_31_0 = (0x1 << 3) | (0x1 << 11);
110 hs_phy->m31_phy_xcfgi_63_32 = 0;
111 hs_phy->m31_phy_xcfgi_95_64 = 0x1 << (72 - 64);
112 hs_phy->m31_phy_xcfgi_127_96 = 0;
113
114 #ifdef USBHS_M31_CLOCK_RECOVERY
115 hs_phy->m31_phy_noncry_rstb = 1;
116 hs_phy->m31_phy_noncry_en = 1;
117 hs_phy->m31_phy_outclksel = 0;
118 hs_phy->m31_phy_coreclkin = 0;
119 hs_phy->m31_phy_xtlsel = 2; /* Select 25 MHz clock */
120 #else
121 hs_phy->m31_phy_noncry_rstb = 0;
122 hs_phy->m31_phy_noncry_en = 0;
123 hs_phy->m31_phy_outclksel = 1;
124 hs_phy->m31_phy_coreclkin = 1;
125 hs_phy->m31_phy_xtlsel = 3; /* Select 30 MHz clock */
126 #endif
127 hs_phy->m31_phy_pll_en = 1;
128 hs_phy->m31_phy_oscouten = 1;
129
130 /* Reset PHY */
131 hs_phy->m31_phy_ponrst = 0;
132 hs_phy->m31_phy_ponrst = 1;
133}
