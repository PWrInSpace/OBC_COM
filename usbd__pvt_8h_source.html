117// Check if endpoint is ready (not busy and not stalled)
118TU_ATTR_ALWAYS_INLINE static inline
119bool usbd_edpt_ready(uint8_t rhport, uint8_t ep_addr) {
120 const bool is_busy = usbd_edpt_busy(rhport, ep_addr);
121 const bool is_stalled = usbd_edpt_stalled(rhport, ep_addr);
122 return !is_busy && !is_stalled;
123}
