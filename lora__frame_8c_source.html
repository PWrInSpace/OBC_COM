59bool lora_frame_validate(const uint8_t *pkt, size_t len, const uint8_t **payload_p, size_t *payload_len_p) {
60 if (pkt == NULL || len < LORA_PACKET_PREFIX_LEN + 2u) return false;
61 if (memcmp(pkt, LORA_PACKET_PREFIX, LORA_PACKET_PREFIX_LEN) != 0) return false;
62
63 size_t pb_len = len - LORA_PACKET_PREFIX_LEN - 1u;
64 uint8_t want = pkt[len - 1u];
65 if (lora_frame_checksum(pkt + LORA_PACKET_PREFIX_LEN, pb_len) != want) return false;
66
67 if (payload_p != NULL) *payload_p = pkt + LORA_PACKET_PREFIX_LEN;
68 if (payload_len_p != NULL) *payload_len_p = pb_len;
69 return true;
70}
