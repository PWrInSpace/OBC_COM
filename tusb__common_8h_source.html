117// This is a backport of memset_s from c11
118TU_ATTR_ALWAYS_INLINE static inline int tu_memset_s(void *dest, size_t destsz, int ch, size_t count) {
119 // Validate parameters
120 if (dest == NULL) {
121 return -1;
122 }
123
124 if (count == 0u) {
125 return 0;
126 }
127
128 if (count > destsz) {
129 return -1;
130 }
131
132 (void) memset(dest, ch, count);
133 return 0;
134}

332// scatter read 4 bytes from two buffers (LE). Parameter are not checked
333TU_ATTR_ALWAYS_INLINE static inline uint32_t tu_scatter_read32(const uint8_t *buf1, uint8_t len1, const uint8_t *buf2,
334 uint8_t len2) {
335 uint32_t result = 0;
336 uint8_t shift = 0;
337
338 for (uint8_t i = 0; i < len1; ++i) {
339 result |= ((uint32_t)buf1[i]) << shift;
340 shift += 8;
341 }
342
343 for (uint8_t i = 0; i < len2; ++i) {
344 result |= ((uint32_t)buf2[i]) << shift;
345 shift += 8;
346 }
347
348 return result;
349}

351// scatter write 4 bytes (LE) to two buffers. Parameter are not checked
352TU_ATTR_ALWAYS_INLINE static inline void tu_scatter_write32(uint32_t value, uint8_t *buf1, uint8_t len1,
353 uint8_t *buf2, uint8_t len2) {
354 for (uint8_t i = 0; i < len1; ++i) {
355 buf1[i] = (uint8_t)(value & 0xFF);
356 value >>= 8;
357 }
358
359 for (uint8_t i = 0; i < len2; ++i) {
360 buf2[i] = (uint8_t)(value & 0xFF);
361 value >>= 8;
362 }
363}
