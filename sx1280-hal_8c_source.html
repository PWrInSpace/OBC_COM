81void SX1280HalClearInstructionRam( SX1280_t *sx1280 )
82{
83 // Clearing the instruction RAM is writing 0x00s on every bytes of the
84 // instruction RAM
85 uint8_t cmd_out[3];
86 cmd_out[0] = RADIO_WRITE_REGISTER;
87 cmd_out[1] = ( IRAM_START_ADDRESS >> 8 ) & 0x00FF;
88 cmd_out[2] = IRAM_START_ADDRESS & 0x00FF;
89
90 uint8_t iram[IRAM_SIZE];
91 memset(iram, 0x00, IRAM_SIZE);
92
93 SX1280HalWaitOnBusy( sx1280 );
94
95 sx1280->spi_write(sx1280->ctx, cmd_out, sizeof(cmd_out), iram, sizeof(iram));
96
97 SX1280HalWaitOnBusy( sx1280 );
98}

118void SX1280HalWriteCommand( SX1280_t *sx1280, uint8_t command, uint8_t *buffer, uint16_t size )
119{
120 SX1280HalWaitOnBusy( sx1280 );
121
122 uint8_t cmd_out[1];
123 cmd_out[0] = command;
124
125 sx1280->spi_write(sx1280->ctx, cmd_out, sizeof(cmd_out), buffer, size);
126
127 if( command != RADIO_SET_SLEEP )
128 {
129 SX1280HalWaitOnBusy( sx1280 );
130 }
131}

218uint8_t SX1280HalGetDioStatus( SX1280_t *sx1280 )
219{
220 uint8_t Status = sx1280->get_busy( sx1280->ctx );
221
222#if( RADIO_DIO1_ENABLE )
223 Status |= (sx1280->get_dio[0](sx1280->ctx) << 1);
224#endif
225#if( RADIO_DIO2_ENABLE )
226 Status |= (sx1280->get_dio[1](sx1280->ctx) << 2);
227#endif
228#if( RADIO_DIO3_ENABLE )
229 Status |= (sx1280->get_dio[2](sx1280->ctx) << 3);
230#endif
231#if( !RADIO_DIO1_ENABLE && !RADIO_DIO2_ENABLE && !RADIO_DIO3_ENABLE )
232#error "Please define a DIO"
233#endif
234
235 return Status;
