36//--------------------------------------------------------------------+
37// Configuration
38//--------------------------------------------------------------------+
39#ifndef CFG_TUD_VENDOR_EPSIZE
40 #define CFG_TUD_VENDOR_EPSIZE 64
41#endif
42
43// RX FIFO can be disabled by setting this value to 0
44#ifndef CFG_TUD_VENDOR_RX_BUFSIZE
45 #define CFG_TUD_VENDOR_RX_BUFSIZE 64
46#endif
47
48// TX FIFO can be disabled by setting this value to 0
49#ifndef CFG_TUD_VENDOR_TX_BUFSIZE
50 #define CFG_TUD_VENDOR_TX_BUFSIZE 64
51#endif
52
53// Vendor is buffered (FIFO mode) if both TX and RX buffers are configured
54// If either is 0, vendor operates in non-buffered (direct transfer) mode
55#ifndef CFG_TUD_VENDOR_TXRX_BUFFERED
56 #define CFG_TUD_VENDOR_TXRX_BUFFERED ((CFG_TUD_VENDOR_RX_BUFSIZE > 0) && (CFG_TUD_VENDOR_TX_BUFSIZE > 0))
57#endif
58
59// Application will manually schedule RX transfer. This can be useful when using with non-fifo (buffered) mode
60// i.e. CFG_TUD_VENDOR_TXRX_BUFFERED = 0
61#ifndef CFG_TUD_VENDOR_RX_MANUAL_XFER
62 #define CFG_TUD_VENDOR_RX_MANUAL_XFER 0
63#endif
64
65//--------------------------------------------------------------------+
66// Application API (Multiple Interfaces) i.e CFG_TUD_VENDOR > 1
67//--------------------------------------------------------------------+
68
69// Return whether the vendor interface is mounted
70bool tud_vendor_n_mounted(uint8_t idx);

72//------------- RX -------------//
73#if CFG_TUD_VENDOR_TXRX_BUFFERED
74// Return number of available bytes for reading
75uint32_t tud_vendor_n_available(uint8_t idx);
76
77// Peek a byte from RX buffer
78bool tud_vendor_n_peek(uint8_t idx, uint8_t *ui8);

80// Read from RX FIFO
81uint32_t tud_vendor_n_read(uint8_t idx, void *buffer, uint32_t bufsize);
82
83// Flush (clear) RX FIFO
84void tud_vendor_n_read_flush(uint8_t idx);
85#endif
86
87#if CFG_TUD_VENDOR_RX_MANUAL_XFER
88// Start a new RX transfer to fill the RX FIFO, return false if previous transfer is still ongoing
89bool tud_vendor_n_read_xfer(uint8_t idx);
90#endif
91
92//------------- TX -------------//
93// Write to TX FIFO. This can be buffered and not sent immediately unless buffered bytes >= USB endpoint size
94uint32_t tud_vendor_n_write(uint8_t idx, const void *buffer, uint32_t bufsize);
95
96// Return number of bytes available for writing in TX FIFO (or endpoint if non-buffered)
97uint32_t tud_vendor_n_write_available(uint8_t idx);
98
99#if CFG_TUD_VENDOR_TXRX_BUFFERED
100// Force sending buffered data, return number of bytes sent
101uint32_t tud_vendor_n_write_flush(uint8_t idx);
102
103// Clear the transmit FIFO
104bool tud_vendor_n_write_clear(uint8_t idx);
105#endif

107// Write a null-terminated string to TX FIFO
108TU_ATTR_ALWAYS_INLINE static inline uint32_t tud_vendor_n_write_str(uint8_t idx, const char *str) {
109 return tud_vendor_n_write(idx, str, strlen(str));
