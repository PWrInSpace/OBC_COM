89static void led_render_pixel(uint16_t i, uint8_t *r, uint8_t *g, uint8_t *b) {
90 *r = 0u;
91 *g = 0u;
92 *b = 0u;
93
94 switch (g_effect[i]) {
95 case LED_EFFECT_SOLID:
96 unpack(g_color[i], r, g, b);
97 break;
98
99 case LED_EFFECT_BLINK:
100 if ((g_frame % (LED_BLINK_ON_FRAMES + LED_BLINK_OFF_FRAMES)) < LED_BLINK_ON_FRAMES) {
101 unpack(g_color[i], r, g, b);
102 }
103 break;
104
105 case LED_EFFECT_RAINBOW:
106 wheel((uint8_t)(g_frame * RAINBOW_HUE_STEP_PER_FRAME + i * RAINBOW_HUE_STEP_PER_LED), r, g, b);
107 break;
108
109 default:
110 break;
111 }
112}

114static void led_cb(void *arg) {
115 (void)arg;
116
117 for (uint16_t i = 0; i < LED_COUNT; i++) {
118 uint8_t r, g, b;
119 led_render_pixel(i, &r, &g, &b);
120 ws2812_set_pixel(&g_ws, i, correct(r), correct(g), correct(b));
121 }
122 g_frame++;
123 ws2812_show(&g_ws);
124}

126void led_init(void) {
127 const ws2812_t cfg = {.htim = &htim4, .channel = TIM_CHANNEL_4, .num_leds = LED_COUNT};
128 ws2812_init(&g_ws, &cfg);
129
130 for (uint16_t i = 0; i < LED_COUNT; i++) {
131 g_effect[i] = LED_EFFECT_OFF;
132 g_color[i] = LED_COLOR_OFF;
133 }
134
135 g_timer = osTimerNew(led_cb, osTimerPeriodic, NULL, NULL);
136 if (g_timer) osTimerStart(g_timer, LED_FRAME_MS);
137}
