36//--------------------------------------------------------------------+
37// Spinlock API
38//--------------------------------------------------------------------+
39// Note: This implementation is designed for bare-metal single-core systems without RTOS.
40// - Supports nested locking within the same execution context
41// - NOT suitable for true SMP (Symmetric Multi-Processing) systems
42// - NOT thread-safe for multi-threaded environments
43// - Primarily manages interrupt enable/disable state for critical sections
44typedef struct {
45 void (* interrupt_set)(bool enabled);
46 uint32_t nested_count;
47} osal_spinlock_t;
48
49// For SMP, spinlock must be locked by hardware, cannot just use interrupt
50#define OSAL_SPINLOCK_DEF(_name, _int_set) \
51 osal_spinlock_t _name = { .interrupt_set = _int_set, .nested_count = 0 }
52
53TU_ATTR_ALWAYS_INLINE static inline void osal_spin_init(osal_spinlock_t *ctx) {
54 (void) ctx;
55}
56
57TU_ATTR_ALWAYS_INLINE static inline void osal_spin_lock(osal_spinlock_t *ctx, bool in_isr) {
58 // Disable interrupts first to make nested_count increment atomic
59 if (!in_isr && ctx->nested_count == 0) {
60 ctx->interrupt_set(false);
61 }
62 ctx->nested_count++;
63}
64
65TU_ATTR_ALWAYS_INLINE static inline void osal_spin_unlock(osal_spinlock_t *ctx, bool in_isr) {
66 if (ctx->nested_count == 0) {
67 return; // spin is not locked to begin with
68 }
69
70 ctx->nested_count--;
71
72 // Only re-enable interrupts when fully unlocked
73 if (!in_isr && ctx->nested_count == 0) {
74 ctx->interrupt_set(true);
75 }
76}

98TU_ATTR_ALWAYS_INLINE static inline bool osal_semaphore_post(osal_semaphore_t sem_hdl, bool in_isr) {
99 (void) in_isr;
100 sem_hdl->count++;
101 return true;
102}

105TU_ATTR_ALWAYS_INLINE static inline bool osal_semaphore_wait(osal_semaphore_t sem_hdl, uint32_t msec) {
106 (void) msec;

109 sem_hdl->count--;

160typedef struct {
161 void (* interrupt_set)(bool enabled);
162 uint16_t item_size;

166typedef osal_queue_def_t* osal_queue_t;

168// _int_set is used as mutex in OS NONE (disable/enable USB ISR)
169#define OSAL_QUEUE_DEF(_int_set, _name, _depth, _type) \
170 uint8_t _name##_buf[_depth * sizeof(_type)]; \
171 osal_queue_def_t _name = {.interrupt_set = _int_set, \
172 .item_size = sizeof(_type), \
173 .ff = TU_FIFO_INIT(_name##_buf, _depth * sizeof(_type), false)}
174
175TU_ATTR_ALWAYS_INLINE static inline osal_queue_t osal_queue_create(osal_queue_def_t* qdef) {
176 tu_fifo_clear(&qdef->ff);
177 return (osal_queue_t) qdef;
178}
179
180TU_ATTR_ALWAYS_INLINE static inline bool osal_queue_delete(osal_queue_t qhdl) {
181 (void) qhdl;
182 return true; // nothing to do
183}

185TU_ATTR_ALWAYS_INLINE static inline bool osal_queue_receive(osal_queue_t qhdl, void* data, uint32_t msec) {
186 (void) msec; // not used, always behave as msec = 0
187
188 qhdl->interrupt_set(false);
189 const bool success = (tu_fifo_read_n(&qhdl->ff, data, qhdl->item_size) > 0);
190 qhdl->interrupt_set(true);
191
192 return success;
193}
194
195TU_ATTR_ALWAYS_INLINE static inline bool osal_queue_send(osal_queue_t qhdl, void const* data, bool in_isr) {
196 if (!in_isr) {
197 qhdl->interrupt_set(false);
198 }
199
200 const bool success = (tu_fifo_write_n(&qhdl->ff, data, qhdl->item_size) > 0);
201
202 if (!in_isr) {
203 qhdl->interrupt_set(true);
204 }
205
206 return success;
207}

209TU_ATTR_ALWAYS_INLINE static inline bool osal_queue_empty(osal_queue_t qhdl) {
210 // Skip queue lock/unlock since this function is primarily called
211 // with interrupt disabled before going into low power mode
212 return tu_fifo_empty(&qhdl->ff);
213}
