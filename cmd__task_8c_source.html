36void cmd_task(void *argument) {
37 (void)argument;
38
39 CMD_Buffer_t *received_ptr = NULL;
40 uint8_t usb_byte;
41 static uint8_t usb_frame_buf[BUFFER_SIZE];
42 static uint16_t usb_idx = 0;
43 uint32_t ulNotifiedValue;
44
45 for(;;) {
46
47 xTaskNotifyWait(0, 0xFFFFFFFF, &ulNotifiedValue, portMAX_DELAY);
48
49 while (xStreamBufferReceive(xUsbStreamBuffer, &usb_byte, 1, 0) > 0) {
50 if (usb_idx == 0 && (usb_byte == '\n' || usb_byte == '\r' || usb_byte == ' ')) {
51 continue;
52 }
53
54 if (usb_idx < BUFFER_SIZE - 1) {
55 usb_frame_buf[usb_idx++] = usb_byte;
56 }
57
58 if (usb_byte == '\n' || usb_byte == '\r') {
59 usb_idx--;
60 usb_frame_buf[usb_idx] = '\0';
61
62 if (usb_idx >= 4 && memcmp(usb_frame_buf, "CMD;", 4) == 0) {
63 process_command(usb_frame_buf, usb_idx);
64 }
65 else if (usb_idx > 0) {
66 USB_Transmit((uint8_t*)"Unknown command format\r\n", 24);
67 lora_gs_tx_enqueue(usb_frame_buf, usb_idx);
68 }
69
70 usb_idx = 0;
71 memset(usb_frame_buf, 0, BUFFER_SIZE);
72 }
73 }
74
75 while (cmd_queue != NULL && xQueueReceive(cmd_queue, &received_ptr, 0) == pdPASS) {
76 if (received_ptr != NULL) {
77 uint16_t actual_len = 0;
78 uint8_t *data = received_ptr->data;
79
80 if (data[0] == 0x32) {
81 // Binary frame detected
82 actual_len = data[2] + 5;
83 if (actual_len > BUFFER_SIZE) actual_len = BUFFER_SIZE;
84
85 USB_Transmit((uint8_t*)"RX BIN CMD (HEX): ", 18);
86 //USB_Transmit_Hex(data, actual_len);
87 } else {
88 // Regular text command
89 actual_len = received_ptr->len;
90 USB_Transmit((uint8_t*)"RX CMD: ", 8);
91 USB_Transmit(data, actual_len);
92 }
93
94 USB_Transmit((uint8_t*)"\r\n", 2);
95 process_command(data, actual_len);
96 memset(data, 0, BUFFER_SIZE);
97 xQueueSend(free_pool_queue, &received_ptr, 0);
98 }
99 }
