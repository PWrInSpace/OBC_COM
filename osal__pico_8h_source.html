128typedef struct {
129 uint16_t item_size;
130 tu_fifo_t ff;
131 struct critical_section critsec; // osal_queue may be used in IRQs, so need critical section
132} osal_queue_def_t;
133
134typedef osal_queue_def_t *osal_queue_t;
135
136// role device/host is used by OS NONE for mutex (disable usb isr) only
137#define OSAL_QUEUE_DEF(_int_set, _name, _depth, _type) \
138 uint8_t _name##_buf[_depth * sizeof(_type)]; \
139 osal_queue_def_t _name = {.item_size = sizeof(_type), .ff = TU_FIFO_INIT(_name##_buf, _depth * sizeof(_type), false)}
140
141TU_ATTR_ALWAYS_INLINE static inline osal_queue_t osal_queue_create(osal_queue_def_t *qdef) {
142 critical_section_init(&qdef->critsec);
143 tu_fifo_clear(&qdef->ff);
144 return (osal_queue_t)qdef;
145}
146
147TU_ATTR_ALWAYS_INLINE static inline bool osal_queue_delete(osal_queue_t qhdl) {
148 osal_queue_def_t *qdef = (osal_queue_def_t *)qhdl;
149 critical_section_deinit(&qdef->critsec);
150 return true;
151}
152
153TU_ATTR_ALWAYS_INLINE static inline bool osal_queue_receive(osal_queue_t qhdl, void *data, uint32_t msec) {
154 (void)msec; // not used, always behave as msec = 0
155
156 critical_section_enter_blocking(&qhdl->critsec);
157 bool success = tu_fifo_read_n(&qhdl->ff, data, qhdl->item_size);
158 critical_section_exit(&qhdl->critsec);
159
160 return success;
161}
162
163TU_ATTR_ALWAYS_INLINE static inline bool osal_queue_send(osal_queue_t qhdl, const void *data, bool in_isr) {
164 (void)in_isr;
165
166 critical_section_enter_blocking(&qhdl->critsec);
167 bool success = tu_fifo_write_n(&qhdl->ff, data, qhdl->item_size);
168 critical_section_exit(&qhdl->critsec);
169
170 return success;
171}
