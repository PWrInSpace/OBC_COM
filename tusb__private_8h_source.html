121// Open an endpoint stream

172// Complete read transfer by writing EP -> FIFO. Must be called in the transfer complete callback
173TU_ATTR_ALWAYS_INLINE static inline
174void tu_edpt_stream_read_xfer_complete(tu_edpt_stream_t* s, uint32_t xferred_bytes) {
175 if (s->ep_buf != NULL) {
176 tu_fifo_write_n(&s->ff, s->ep_buf, (uint16_t)xferred_bytes);
177 }
178}

184}
