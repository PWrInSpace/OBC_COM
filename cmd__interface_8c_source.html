108void process_text_packet(char *raw_str) {
109 strtok(raw_str, ";"); // Skip "CMD"
110 char *cmd_str = strtok(NULL, ";");
111 char *val_str = strtok(NULL, ";");
112
113 if (cmd_str == NULL) return;
114
115 for (size_t i = 0; i < cmd_map_size; i++) {
116 if (strcmp(cmd_str, cmd_map[i].name) == 0) {
117 cmd_params_t p = {
118 .data = (uint8_t*)val_str,
119 .len = (val_str ? (uint16_t)strlen(val_str) : 0),
120 .is_binary = false
121 };
122 cmd_map[i].handler(&p);
123 return;
124 }
125 }
126}

128void process_binary_packet(uint8_t *buf, uint16_t len) {
129 if (len < 3) return;
130
131 uint8_t cmd_id = buf[1];
132 uint8_t data_len = buf[2];
133
134 for (size_t i = 0; i < cmd_map_size; i++) {
135 if (cmd_map[i].id == (Command_t)cmd_id) {
136 cmd_params_t p = {
137 .data = (data_len > 0) ? &buf[3] : NULL,
138 .len = data_len,
139 .is_binary = true
140 };
141 cmd_map[i].handler(&p);
142 return;
143 }
144 }

147void process_command(uint8_t *rx_buf, uint16_t len) {
148 if (len == 0) return;
149
150 if (rx_buf[0] == 0x32 && len >= 5) {
151 process_binary_packet(rx_buf, len);
152 }
153 else if (len >= 4 && memcmp(rx_buf, "CMD;", 4) == 0) {
154 if (len < MAX_CMD_LEN) {
155 rx_buf[len] = '\0';
156 process_text_packet((char*)rx_buf);
157 }
158 }

161void handle_logdump(cmd_params_t *params) {
162 (void)params;
163 const char *hdr = "\r\n--- LOG DUMP ---\r\n";
164 const char *ftr = "\r\n--- END DUMP ---\r\n";
165 const char *buf = logger_get_buffer();
166 uint32_t pos = logger_get_write_position();
167
168 USB_Transmit((uint8_t*)hdr, (uint16_t)strlen(hdr));
169 for (uint32_t i = 0; i < pos; ) {
170 uint16_t chunk = (uint16_t)((pos - i > 256u) ? 256u : (pos - i));
171 USB_Transmit((uint8_t*)(buf + i), chunk);
172 i += chunk;
173 }
174 USB_Transmit((uint8_t*)ftr, (uint16_t)strlen(ftr));
175}

345void handle_log_mute(cmd_params_t *params) {
346 const char* tag_str = (char*)params->data;
347 uint32_t bit = logger_get_bit_from_tag(tag_str);
348
349 if (bit == 0) {
350 USB_Transmit((uint8_t*)"ERR: Unknown TAG\r\n", 18);
351 return;
352 }
353 current_log_mute_mask ^= bit;
354 NVS_Write(PARAM_LOG_TAG_MUTE, current_log_mute_mask);
355
356 char resp[64];
357 bool muted = (current_log_mute_mask & bit);
358 snprintf(resp, sizeof(resp), "LOG: %s is now %s\r\n", tag_str, muted ? "MUTED" : "ACTIVE");
359 USB_Transmit((uint8_t*)resp, strlen(resp));
360}
