96 .PacketType = PACKET_TYPE_LORA,
97 .Params.LoRa = {

138void rfm95_print_actual_settings(rfm95_t *rfm) {
139 char buf[128];
140
141 // 1. Read Registers
142 uint8_t version = rfm95_read_reg(rfm, 0x42);
143 uint8_t opMode = rfm95_read_reg(rfm, 0x01);
144 uint8_t pa_config = rfm95_read_reg(rfm, 0x09);
145 uint8_t mc1 = rfm95_read_reg(rfm, 0x1D);
146 uint8_t mc2 = rfm95_read_reg(rfm, 0x1E);
147 uint8_t sync_word = rfm95_read_reg(rfm, REG_SYNC_WORD);
148 uint16_t preamble = ((uint16_t)rfm95_read_reg(rfm, REG_PREAMBLE_MSB) << 8) |
149 (uint16_t)rfm95_read_reg(rfm, REG_PREAMBLE_LSB);
150
151 static const char *bw_khz_tbl[10] = {
152 "7.8", "10.4", "15.6", "20.8", "31.25", "41.7", "62.5", "125", "250", "500"
153 };
154 uint8_t sf_i = (mc2 >> 4) & 0x0F; /* Spreading Factor */
155 uint8_t bw_i = (mc1 >> 4) & 0x0F; /* BW index */
156 const char *bw_s = (bw_i < 10) ? bw_khz_tbl[bw_i] : "?";
157 uint8_t cr_i = (mc1 >> 1) & 0x07; /* Coding Rate: 1->4/5 .. 4->4/8 */
158 uint8_t implicit = (mc1 & 0x01); /* 1 = implicit header */
159 uint8_t crc_on = (mc2 >> 2) & 0x01; /* 1 = CRC on */
160
161 // 2. Power Decoding (Integer Math)
162 uint8_t pa_select = (pa_config >> 7) & 0x01;
163 uint8_t max_pwr_i = (pa_config >> 4) & 0x07;
164 uint8_t out_pwr_i = (pa_config & 0x0F);
165
166 // Pmax = 10.8 + 0.6 * MaxPower -> working in 1/10th dBs
167 // 108 + (6 * max_pwr_i)
168 int32_t p_max_x10 = 108 + (6 * max_pwr_i);
169 int32_t p_out_x10 = 0;
170
171 if (pa_select) {
172 // PA_BOOST: Pout = 17 - (15 - OutputPower)
173 // In tenths: 170 - (150 - (out_pwr_i * 10))
174 p_out_x10 = 170 - (150 - (out_pwr_i * 10));
175 } else {
176 // RFO: Pout = Pmax - (15 - OutputPower)
177 p_out_x10 = p_max_x10 - (150 - (out_pwr_i * 10));
178 }
179
180 // 3. Frequency Calculation
181 uint32_t frf = ((uint32_t)rfm95_read_reg(rfm, 0x06) << 16) |
182 ((uint32_t)rfm95_read_reg(rfm, 0x07) << 8) |
183 ((uint32_t)rfm95_read_reg(rfm, 0x08));
184 uint32_t freq_hz = (uint32_t)((uint64_t)frf * 32000000 / 524288);
185
186 // --- 4. Print Table ---
187 USB_Transmit((uint8_t*)"\r\n+-----------------------+-----------------------+\r\n", 51);
188 USB_Transmit((uint8_t*)"| RFM95 HARDWARE DEBUG | VALUE |\r\n", 51);
189 USB_Transmit((uint8_t*)"+-----------------------+-----------------------+\r\n", 51);
190
191 sprintf(buf, "| Silicon Version | 0x%02X (Expect 0x12) |\r\n", version);
192 USB_Transmit((uint8_t*)buf, strlen(buf));
193
194 sprintf(buf, "| PA_BOOST Pin | %-21s |\r\n", pa_select ? "ON" : "OFF (RFO Pin)");
195 USB_Transmit((uint8_t*)buf, strlen(buf));
196
197 // Splitting float into two ints: p_max_x10 / 10 and p_max_x10 % 10
198 sprintf(buf, "| MaxPower (Pmax) | %2ld.%1ld dBm (Val: %d) |\r\n",
199 p_max_x10 / 10, p_max_x10 % 10, max_pwr_i);
200 USB_Transmit((uint8_t*)buf, strlen(buf));
201
202 sprintf(buf, "| Calculated Pout | %2ld.%1d dBm |\r\n",
203 p_out_x10 / 10, abs(p_out_x10 % 10));
204 USB_Transmit((uint8_t*)buf, strlen(buf));
205
206 sprintf(buf, "| Frequency | %-18lu Hz |\r\n", freq_hz);
207 USB_Transmit((uint8_t*)buf, strlen(buf));
208
209 sprintf(buf, "| Spreading Factor | SF%-19d |\r\n", sf_i);
210 USB_Transmit((uint8_t*)buf, strlen(buf));
211
212 sprintf(buf, "| Bandwidth | %s kHz (idx %d)%*s |\r\n",
213 bw_s, bw_i, (int)(9 - strlen(bw_s)), "");
214 USB_Transmit((uint8_t*)buf, strlen(buf));
215
216 sprintf(buf, "| Coding Rate | 4/%-19d |\r\n", cr_i + 4);
217 USB_Transmit((uint8_t*)buf, strlen(buf));
218
219 sprintf(buf, "| CRC | %-21s |\r\n", crc_on ? "ON" : "OFF");
220 USB_Transmit((uint8_t*)buf, strlen(buf));
221
222 sprintf(buf, "| Header | %-21s |\r\n", implicit ? "IMPLICIT" : "EXPLICIT");
223 USB_Transmit((uint8_t*)buf, strlen(buf));
224
225 sprintf(buf, "| Sync Word (0x39) | 0x%02X%-17s |\r\n", sync_word, "");
226 USB_Transmit((uint8_t*)buf, strlen(buf));
227
228 sprintf(buf, "| Preamble Length | %-21u |\r\n", preamble);
229 USB_Transmit((uint8_t*)buf, strlen(buf));
230
231 // Raw Register Dump
232 USB_Transmit((uint8_t*)"+-----------------------+-----------------------+\r\n", 51);
233 USB_Transmit((uint8_t*)"| REGISTER ADDRESS | HEX VALUE |\r\n", 51);
234 USB_Transmit((uint8_t*)"+-----------------------+-----------------------+\r\n", 51);
235
236 sprintf(buf, "| RegOpMode (0x01) | 0x%02X |\r\n", opMode);
237 USB_Transmit((uint8_t*)buf, strlen(buf));
238
239 sprintf(buf, "| RegPaConfig (0x09) | 0x%02X |\r\n", pa_config);
240 USB_Transmit((uint8_t*)buf, strlen(buf));
241
242 USB_Transmit((uint8_t*)"+-----------------------+-----------------------+\r\n", 51);
243}

245uint8_t rfm95w_read_status(rfm95_t *rfm)
246{
247 uint8_t opMode = rfm95_read_reg(rfm, 0x01);
248 opMode = opMode&0x03;
249 switch (opMode)
250 {
251 case 0x00:
252 USB_Transmit((uint8_t*)"Status: Sleep\r\n", 16);
253 break;
254 case 0x01:
255 USB_Transmit((uint8_t*)"Status: Standby\r\n", 18);
256 break;
257 case 0x02:
258 USB_Transmit((uint8_t*)"Status: Frequency Synthesizer TX\r\n", 35);
259 break;
260 case 0x03:
261 USB_Transmit((uint8_t*)"Status: Transmit\r\n", 18);
262 break;
263 case 0x04:
264 USB_Transmit((uint8_t*)"Status: Frequency Synthesizer RX\r\n", 35);
265 break;
266 case 0x05:
267 USB_Transmit((uint8_t*)"Status: Receive\r\n", 18);
268 break;
269 }
270 return opMode;
271}
