65static void snd_cb(void *arg) {
66 (void)arg;
67 note_t n;
68 bool play;
69
70 taskENTER_CRITICAL();
71 play = (g_seq != NULL) && (g_idx < g_len);
72 if (play) n = g_seq[g_idx++];
73 taskEXIT_CRITICAL();
74
75 if (!play) {
76 buzzer_mute(&g_buzzer);
77 return;
78 }
79
80 if (n.freq_hz != 0u)
81 buzzer_set_tone(&g_buzzer, n.freq_hz);
82 else
83 buzzer_mute(&g_buzzer);
84
85 osTimerStart(g_timer, pdMS_TO_TICKS(n.ms != 0u ? n.ms : 1u));
86}

101void sound_play(sound_id_t id) {
102 if (id >= SOUND_COUNT || g_timer == NULL) return;
103
104 const melody_t *m = &g_melodies[id];
105 if (m->seq == NULL || m->len == 0u) return;
106
107 taskENTER_CRITICAL();
108 g_seq = m->seq;
109 g_len = m->len;
110 g_idx = 0u;
111 taskEXIT_CRITICAL();
112
113 osTimerStart(g_timer, 1u);
114}
