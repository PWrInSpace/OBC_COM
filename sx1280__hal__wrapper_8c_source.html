27int32_t sx1280_spi_write_dma(void* ctx, uint8_t *prefix, uint16_t prefix_len, uint8_t* out, uint16_t out_len) {
28 SPI_HandleTypeDef* hspi = (SPI_HandleTypeDef*)ctx;
29 uint16_t total_len = prefix_len + out_len;
30
31 if (xSemaphoreTake(sx1280_spi_sem, pdMS_TO_TICKS(100)) != pdTRUE) return -1;
32 if (total_len > MAX_SPI_BUFFER) { xSemaphoreGive(sx1280_spi_sem); return -2; }
33
34 // while(HAL_GPIO_ReadPin(BUSY_SX1280_GPIO_Port, BUSY_SX1280_Pin) == GPIO_PIN_SET);
35
36 memcpy(spi_dma_tx_buffer, prefix, prefix_len);
37 if (out != NULL && out_len > 0) {
38 memcpy(spi_dma_tx_buffer + prefix_len, out, out_len);
39 }
40
41 HAL_GPIO_WritePin(SX1280_CS_GPIO_Port, SX1280_CS_Pin, GPIO_PIN_RESET);
42
43 if (HAL_SPI_Transmit_DMA(hspi, spi_dma_tx_buffer, total_len) != HAL_OK) {
44 HAL_GPIO_WritePin(SX1280_CS_GPIO_Port, SX1280_CS_Pin, GPIO_PIN_SET);
45 xSemaphoreGive(sx1280_spi_sem);
46 return -3;
47 }
48 return 0;
49}

51int32_t sx1280_spi_read_dma(void* ctx, uint8_t *prefix, uint16_t prefix_len, uint8_t* in, uint16_t in_len) {
52 SPI_HandleTypeDef* hspi = (SPI_HandleTypeDef*)ctx;
53
54 if (xSemaphoreTake(sx1280_spi_sem, pdMS_TO_TICKS(100)) != pdTRUE) return -1;
55
56 // while(HAL_GPIO_ReadPin(BUSY_SX1280_GPIO_Port, BUSY_SX1280_Pin) == GPIO_PIN_SET);
57
58 HAL_GPIO_WritePin(SX1280_CS_GPIO_Port, SX1280_CS_Pin, GPIO_PIN_RESET);
59 HAL_SPI_Transmit(hspi, prefix, prefix_len, HAL_MAX_DELAY);
60
61 if (in != NULL && in_len > 0) {
62 if (HAL_SPI_Receive_DMA(hspi, in, in_len) != HAL_OK) {
63 HAL_GPIO_WritePin(SX1280_CS_GPIO_Port, SX1280_CS_Pin, GPIO_PIN_SET);
64 xSemaphoreGive(sx1280_spi_sem);
65 return -3;
66 }
67 } else {
68 HAL_GPIO_WritePin(SX1280_CS_GPIO_Port, SX1280_CS_Pin, GPIO_PIN_SET);
69 xSemaphoreGive(sx1280_spi_sem);
70 }
