13static void forward_window(const gs_radio_iface_t *io, uint32_t duration_ms) {
14 uint8_t buf[LORA_FRAME_MAX];
15 uint32_t start = io->now_ms(io->ctx);
16
17 io->enter_rx(io->ctx);
18 while (elapsed_since(io, start) < duration_ms) {
19 size_t n = io->poll_rx(io->ctx, buf, sizeof(buf));
20 if (n > 0U) {
