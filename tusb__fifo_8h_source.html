54// Due to the use of unmasked pointers, this FIFO does not suffer from losing
55// one item slice. Furthermore, write and read operations are completely
56// decoupled as write and read functions do not modify a common state. Henceforth,
57// writing or reading from the FIFO within an ISR is safe as long as no other
58// process (thread or ISR) interferes.
59// Also, this FIFO is ready to be used in combination with a DMA as the write and
60// read pointers can be updated from within a DMA ISR. Overflows are detectable
61// within a certain number (see tu_fifo_overflow()).
62
63/* Write/Read "pointer" is in the range of: 0 .. depth - 1, and is used to get the fifo data.
64 * Write/Read "index" is always in the range of: 0 .. 2*depth-1
65 *
66 * The extra window allow us to determine the fifo state of empty or full with only 2 indices
67 * Following are examples with depth = 3
68 *
69 * - empty: W = R
70 * |
71 * -------------------------
72 * | 0 | RW| 2 | 3 | 4 | 5 |
73 *
74 * - full 1: W > R
75 * |
76 * -------------------------
77 * | 0 | R | 2 | 3 | W | 5 |
78 *
79 * - full 2: W < R
80 * |
81 * -------------------------
82 * | 0 | 1 | W | 3 | 4 | R |
83 *
84 * - Number of items in the fifo can be determined in either cases:
85 * - case W >= R: Count = W - R
86 * - case W < R: Count = 2*depth - (R - W)
87 *
88 * In non-overwritable mode, computed Count (in above 2 cases) is at most equal to depth.
89 * However, in over-writable mode, write index can be repeatedly increased and count can be
90 * temporarily larger than depth (overflowed condition) e.g
91 *
92 * - Overflowed 1: write(3), write(1)
93 * In this case we will adjust Read index when read()/peek() is called so that count = depth.
94 * |
95 * -------------------------
96 * | R | 1 | 2 | 3 | W | 5 |
97 *
98 * - Double Overflowed i.e index is out of allowed range [0,2*depth)
99 * This occurs when we continue to write after 1st overflowed to 2nd overflowed. e.g:
100 * write(3), write(1), write(2)
101 * This must be prevented since it will cause unrecoverable state, in above example
102 * if not handled the fifo will be empty instead of continue-to-be full. Since we must not modify
103 * read index in write() function, which cause race condition. We will re-position write index so that
104 * after data is written it is a full fifo i.e W = depth - R
105 *
106 * re-position W = 1 before write(2)
107 * Note: we should also move data from mem[3] to read index as well, but deliberately skipped here
108 * since it is an expensive operation !!!
109 * |
110 * -------------------------
111 * | R | W | 2 | 3 | 4 | 5 |
112 *
113 * perform write(2), result is still a full fifo.
114 *
115 * |
116 * -------------------------
117 * | R | 1 | 2 | W | 4 | 5 |
118 */

257//--------------------------------------------------------------------+
258// Internal Helper Local
259// work on local copies of read/write indices in order to only access them once for re-entrancy
260//--------------------------------------------------------------------+
261// return overflowable count (index difference), which can be used to determine both fifo count and an overflow state
262TU_ATTR_ALWAYS_INLINE static inline uint16_t tu_ff_overflow_count(uint16_t depth, uint16_t wr_idx, uint16_t rd_idx) {
263 const int32_t diff = (int32_t)wr_idx - (int32_t)rd_idx;
264 if (diff >= 0) {
265 return (uint16_t)diff;
266 } else {
267 return (uint16_t)(2 * depth + diff);
268 }
269}
270
271// return remaining slot in fifo
272TU_ATTR_ALWAYS_INLINE static inline uint16_t tu_ff_remaining_local(uint16_t depth, uint16_t wr_idx, uint16_t rd_idx) {
273 const uint16_t ovf_count = tu_ff_overflow_count(depth, wr_idx, rd_idx);
274 return (depth > ovf_count) ? (depth - ovf_count) : 0;
275}

286TU_ATTR_ALWAYS_INLINE static inline bool tu_fifo_empty(const tu_fifo_t *f) {
287 const uint16_t wr_idx = f->wr_idx;
288 const uint16_t rd_idx = f->rd_idx;

292// return number of items in fifo, capped to fifo's depth
293TU_ATTR_ALWAYS_INLINE static inline uint16_t tu_fifo_count(const tu_fifo_t *f) {
294 const uint16_t wr_idx = f->wr_idx;
295 const uint16_t rd_idx = f->rd_idx;
296 return tu_min16(tu_ff_overflow_count(f->depth, wr_idx, rd_idx), f->depth);
297}
298
299// check if fifo is full
300TU_ATTR_ALWAYS_INLINE static inline bool tu_fifo_full(const tu_fifo_t *f) {
301 const uint16_t wr_idx = f->wr_idx;
302 const uint16_t rd_idx = f->rd_idx;
303 return tu_ff_overflow_count(f->depth, wr_idx, rd_idx) >= f->depth;
304}

306TU_ATTR_ALWAYS_INLINE static inline uint16_t tu_fifo_remaining(const tu_fifo_t *f) {
307 const uint16_t wr_idx = f->wr_idx;
308 const uint16_t rd_idx = f->rd_idx;
309 return tu_ff_remaining_local(f->depth, wr_idx, rd_idx);
310}
