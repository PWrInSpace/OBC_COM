57static bool wait_for_tx_done(SX1280_t *radio, uint32_t timeout_ms) {
58 uint32_t t0 = HAL_GetTick();
59 while (((SX1280GetIrqStatus(radio) & IRQ_TX_DONE) != IRQ_TX_DONE) &&
60 ((HAL_GetTick() - t0) < timeout_ms)) {
61 osDelay(1);
62 }
63 return ((SX1280GetIrqStatus(radio) & IRQ_TX_DONE) == IRQ_TX_DONE);
64}

77static bool handle_rx_done_and_get_payload(SX1280_t *radio, uint8_t *out_buf, uint8_t *out_size) {
78 if (out_buf == NULL || out_size == NULL) return false;

92static bool rx_wait_for_event(SX1280_t *radio, uint32_t ceiling_ms, uint8_t *out_buf, uint8_t *out_size) {
93
94 uint32_t start = HAL_GetTick();
95 if (out_buf == NULL || out_size == NULL) return false;
96 *out_size = 0;
97
98 for (;;) {
99 uint16_t irq = SX1280GetIrqStatus(radio);
100 if ((irq & IRQ_RX_DONE) == IRQ_RX_DONE) {
101 if (handle_rx_done_and_get_payload(radio, out_buf, out_size)) return true;
102 return false;
103 }
104 if ((irq & IRQ_RX_TX_TIMEOUT) == IRQ_RX_TX_TIMEOUT) {
105 clear_irqs_and_standby(radio);
106 return false;
107 }
108 if ((HAL_GetTick() - start) >= (ceiling_ms + 10)) {
109 /* safety fallback */
110 clear_irqs_and_standby(radio);
111 return false;
112 }
113 osDelay(5);
114 }
115}

117static void send_window(SX1280_t *radio, const uint8_t *payload, uint8_t payload_len, uint32_t window_ms) {
118 SX1280ClearIrqStatus(radio, IRQ_RADIO_ALL);
119 uint32_t start = HAL_GetTick();
120 while ((HAL_GetTick() - start) < window_ms) {
121 SX1280SendPayload(radio, (uint8_t*)payload, payload_len, RX_TX_SINGLE);
122 /* wait for TX_DONE (short timeout) */
123 wait_for_tx_done(radio, TX_DONE_TIMEOUT_MS_DEFAULT);
124 SX1280ClearIrqStatus(radio, IRQ_RADIO_ALL);
125 osDelay(1);
126 }
127 SX1280SetStandby(radio, STDBY_RC);
128}

130static void sx1280_recv_once_ceiling(SX1280_t *radio, uint32_t ceiling_ms) {
131 uint8_t rx_buf[255];
132 uint8_t rx_size = 0;
133
134 /* Start RX using helper */
135 start_rx(radio, ceiling_ms);
136
137 /* Wait for event and fill buffer */
138 if (rx_wait_for_event(radio, ceiling_ms, rx_buf, &rx_size)) {
139 /* process received frame */
140 if (rx_size > 0) {
141 parse_frame(rx_buf, rx_size);
142 }
143 } else {
144 /* no frame received or timeout - ensure radio is in standby */
145 stop_radio_standby(radio);
146 }
