34EE_Status NVS_Write(uint16_t virt_addr, uint32_t data) {
35 EE_Status status = 1;
36
37 if (xSemaphoreTake(xNvsMutex, portMAX_DELAY) == pdTRUE) {
38 status = EE_WriteVariable32bits(virt_addr, data);
39
40 /* If flash pages are full, the library requires a 'Transfer' (Cleanup) */
41 if ((status & EE_STATUSMASK_CLEANUP) == EE_STATUSMASK_CLEANUP) {
42 status = EE_CleanUp();
43 }
44 xSemaphoreGive(xNvsMutex);
45 }
46 HAL_Delay(10); // wait for save the data
47 return status;
48}
