50void rfm95_send_window(rfm95_t *radio, const uint8_t *payload, uint8_t payload_len, uint32_t window_ms) {
51 (void)window_ms;
52 rfm95_write_reg(radio, REG_IRQ_FLAGS, IRQ_ALL);
53 rfm95_set_transmit_mode(radio);
54 rfm95_send_packet(radio, (uint8_t *)payload, payload_len);
55
56 uint32_t start = HAL_GetTick();
57 while (!rfm95_check_tx_done(radio)) {
58 if ((HAL_GetTick() - start) > 5U) {
59 break;
60 }
61 osDelay(1);
62 }
63
64 rfm95_write_reg(radio, REG_IRQ_FLAGS, IRQ_ALL);
65 rfm95_idle(radio);
66}

79static size_t gs_poll_rx(void *ctx, uint8_t *buf, size_t cap) {
80 rfm95_t *radio = (rfm95_t *)ctx;
81
82 uint8_t irq = rfm95_read_reg(radio, REG_IRQ_FLAGS);
83 if (!(irq & IRQ_RX_DONE_MASK)) {
84 osDelay(1);
85 return 0;
86 }
87
88 uint8_t max = (cap > 255U) ? 255U : (uint8_t)cap;
89 uint8_t n = rfm95_receive_packet(radio, buf, max);
90 rfm95_write_reg(radio, REG_IRQ_FLAGS, IRQ_ALL);
91 if (n == 0U) {
92 osDelay(1);
93 return 0;
94 }
95
96 int16_t rssi = rfm95_packet_rssi(radio);
97 if (g_state_mutex != NULL && xSemaphoreTake(g_state_mutex, 0) == pdTRUE) {
98 g_system_state.RSSI = rssi;
99 xSemaphoreGive(g_state_mutex);
100 }
101 HAL_GPIO_TogglePin(RX_RFM_GPIO_Port, RX_RFM_Pin);
102 LOG_INFO("RX len=%u rssi=%d", (unsigned)n, (int)rssi);
103 return n;
104}

124void rfm95wTaskEntry(void *argument) {
125 (void)argument;
126
127 rfm95_t *radio = get_lora_devs_instance()->rfm95w;
128 lora_txq_init(&s_txq);
129
130 rfm95w_config_init();
131 osDelay(2000);
132 nvs_get_rfm95_settings(radio);
133 osDelay(100);
134 rfm95w_config_init_param();
135
136 gs_radio_iface_t io = {
137 .now_ms = gs_now_ms,
138 .enter_rx = gs_enter_rx,
139 .poll_rx = gs_poll_rx,
140 .send = gs_send,
141 .forward = gs_forward,
142 .ctx = radio,
143 .txq = &s_txq,
144 };
145
146 gs_enter_rx(radio);
147
148 for (;;) {
149 if (s_settings_dirty) {
150 s_settings_dirty = false;
151 LOG_INFO("GS RFM95W: reloading radio settings from NVS");
152 apply_nvs_settings(radio);
153 }
154
155 gs_run_cycle(&io);
156 }
157}
