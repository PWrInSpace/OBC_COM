91//--------------------------------------------------------------------+
92// Control Endpoint API
93//--------------------------------------------------------------------+
94
95// Get current protocol: HID_PROTOCOL_BOOT (0) or HID_PROTOCOL_REPORT (1)
96// Note: Device will be initialized in Boot protocol for simplicity.
97// Application can use set_protocol() to switch back to Report protocol.
98uint8_t tuh_hid_get_protocol(uint8_t dev_addr, uint8_t idx);
99
100// Device by default is enumerated in Boot protocol for simplicity. Application
101// can use this to modify the default protocol for next enumeration.
102void tuh_hid_set_default_protocol(uint8_t protocol);
103
104// Set protocol to HID_PROTOCOL_BOOT (0) or HID_PROTOCOL_REPORT (1)
105// This function is only supported by Boot interface (tuh_n_hid_interface_protocol() != NONE)
106bool tuh_hid_set_protocol(uint8_t dev_addr, uint8_t idx, uint8_t protocol);
107
108// Get Report using control endpoint
109// report_type is either Input, Output or Feature, (value from hid_report_type_t)
110bool tuh_hid_get_report(uint8_t dev_addr, uint8_t idx, uint8_t report_id, uint8_t report_type, void *report,
111 uint16_t len);
112
113// Set Report using control endpoint
114// report_type is either Input, Output or Feature, (value from hid_report_type_t)
115bool tuh_hid_set_report(uint8_t dev_addr, uint8_t idx, uint8_t report_id, uint8_t report_type, void *report,
116 uint16_t len);
117
118//--------------------------------------------------------------------+
119// Interrupt Endpoint API
120//--------------------------------------------------------------------+
121
122// Check if HID interface is ready to receive report
123bool tuh_hid_receive_ready(uint8_t dev_addr, uint8_t idx);
124
125// Try to receive next report on Interrupt Endpoint. Immediately return
126// - true If succeeded, tuh_hid_report_received_cb() callback will be invoked when report is available
127// - false if failed to queue the transfer e.g endpoint is busy
128bool tuh_hid_receive_report(uint8_t dev_addr, uint8_t idx);
129
130// Abort receiving report on Interrupt Endpoint
131bool tuh_hid_receive_abort(uint8_t dev_addr, uint8_t idx);
132
133// Check if HID interface is ready to send report
134bool tuh_hid_send_ready(uint8_t dev_addr, uint8_t idx);
135
136// Send report using interrupt endpoint
137// If report_id > 0 (composite), it will be sent as 1st byte, then report contents. Otherwise only report content is
138// sent.
139bool tuh_hid_send_report(uint8_t dev_addr, uint8_t idx, uint8_t report_id, const void *report, uint16_t len);
140
141//--------------------------------------------------------------------+
142// Callbacks (Weak is optional)
143//--------------------------------------------------------------------+
144
145// Invoked when device with hid interface is mounted
146// Report descriptor is also available for use. tuh_hid_parse_report_descriptor()
147// can be used to parse common/simple enough descriptor.
148// Note: if report descriptor length > CFG_TUH_ENUMERATION_BUFSIZE, it will be skipped
149// therefore report_desc = NULL, desc_len = 0
150void tuh_hid_mount_cb(uint8_t dev_addr, uint8_t idx, const uint8_t *report_desc, uint16_t desc_len);
151
152// Invoked when device with hid interface is un-mounted
153void tuh_hid_umount_cb(uint8_t dev_addr, uint8_t idx);
154
155// Invoked when received report from device via interrupt endpoint
156// Note: if there is report ID (composite), it is 1st byte of report
157void tuh_hid_report_received_cb(uint8_t dev_addr, uint8_t idx, const uint8_t *report, uint16_t len);
158
159// Invoked when sent report to device successfully via interrupt endpoint
160void tuh_hid_report_sent_cb(uint8_t dev_addr, uint8_t idx, const uint8_t *report, uint16_t len);
161
162// Invoked when Get Report to device via either control endpoint
163// len = 0 indicate there is error in the transfer e.g stalled response
164void tuh_hid_get_report_complete_cb(uint8_t dev_addr, uint8_t idx, uint8_t report_id, uint8_t report_type,
165 uint16_t len);
166
167// Invoked when Sent Report to device via either control endpoint
168// len = 0 indicate there is error in the transfer e.g stalled response
169void tuh_hid_set_report_complete_cb(uint8_t dev_addr, uint8_t idx, uint8_t report_id, uint8_t report_type,
170 uint16_t len);
171
172// Invoked when Set Protocol request is complete
173void tuh_hid_set_protocol_complete_cb(uint8_t dev_addr, uint8_t idx, uint8_t protocol);
