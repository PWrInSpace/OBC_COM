45uint8_t GPS_AlignBuffer(GNSS_StateHandle *GNSS) {
46 int found_at = -1;
47 uint8_t temp[100];
48 const uint16_t BUF_SIZE = 256;
49
50 for (int i = 0; i < (BUF_SIZE - 4); i++) {
51 if (GNSS->uartWorkingBuffer[i] == 0xB5 &&
52 GNSS->uartWorkingBuffer[i + 1] == 0x62 &&
53 GNSS->uartWorkingBuffer[i + 2] == 0x01 &&
54 GNSS->uartWorkingBuffer[i + 3] == 0x07) {
55 found_at = i;
56 break;
57 }
58 }
59
60 if (found_at != -1) {
61 for (int j = 0; j < 100; j++) {
62 temp[j] = GNSS->uartWorkingBuffer[(found_at + j) % BUF_SIZE];
63 }
64 memcpy(GNSS->uartWorkingBuffer, temp, 100);
65 return 1;
66 }
67 return 0;
68}

86void configure_gps() {
87 uint8_t dummy;
88
89 HAL_GPIO_WritePin(GPS_RST_GPIO_Port, GPS_RST_Pin, GPIO_PIN_RESET);
90 HAL_Delay(50);
91 HAL_GPIO_WritePin(GPS_RST_GPIO_Port, GPS_RST_Pin, GPIO_PIN_SET);
92 osDelay(200);
93
94 while(HAL_UART_Receive(&huart1, &dummy, 1, 5) == HAL_OK);

97 osDelay(50);
98 HAL_UART_Transmit(&huart1, (uint8_t*)disableInfMessages, sizeof(disableInfMessages), 100);
99 osDelay(50);
100 HAL_UART_Transmit(&huart1, (uint8_t*)disableNmeaAll, sizeof(disableNmeaAll), 100);
101 osDelay(50);
102 HAL_UART_Transmit(&huart1, (uint8_t*)setRocketMode4G, sizeof(setRocketMode4G), 100);
103 osDelay(50);
104 HAL_UART_Transmit(&huart1, (uint8_t*)enableNavPvt, sizeof(enableNavPvt), 100);
105 osDelay(50);
106 HAL_UART_Transmit(&huart1, (uint8_t*)setRate1Hz_M10, sizeof(setRate1Hz_M10), 100);
107 osDelay(100);
108
109 HAL_UARTEx_ReceiveToIdle_DMA(&huart1, gpsHandle.uartWorkingBuffer, 256);
110}

112void gps_task(void *argument) {
113 (void)argument;
114
115 configure_gps();
116
117 for(;;) {
118
119 if (ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(1100)) > 0) {
120
121 if (GPS_AlignBuffer(&gpsHandle)) {
122 GNSS_ParsePVTData(&gpsHandle);
123
124 if (xSemaphoreTake(g_state_mutex, portMAX_DELAY) == pdTRUE) {
125 g_system_state.lat = gpsHandle.lat;
126 g_system_state.lon = gpsHandle.lon;
127 g_system_state.hMSL = gpsHandle.hMSL;
128 g_system_state.gSpeed = gpsHandle.gSpeed;
129 g_system_state.numSV = gpsHandle.numSV;
130 g_system_state.fixType = gpsHandle.fixType;
131 xSemaphoreGive(g_state_mutex);
132
133 }
134
135 GPS_PrintStatus(&gpsHandle);
136}
137
138 HAL_UART_AbortReceive(&huart1);
139 __HAL_UART_SEND_REQ(&huart1, UART_RXDATA_FLUSH_REQUEST);
140 HAL_UARTEx_ReceiveToIdle_DMA(&huart1, gpsHandle.uartWorkingBuffer, 256);
141
142 } else {
143 USB_Transmit((uint8_t*)"OBC_GPS: TIMEOUT\r\n", 18);
144 HAL_UART_AbortReceive(&huart1);
145 HAL_UARTEx_ReceiveToIdle_DMA(&huart1, gpsHandle.uartWorkingBuffer, 256);
146 }
147 }
