84bool sd_mount(void) {
85 if (is_mounted) return true;
86
87 osMutexAcquire(sd_mutex_id, osWaitForever);
88 FRESULT res = f_mount(&fs, SDPath, 1);
89 if (res != FR_OK) {
90 LOG_ERROR("sd_mount: f_mount res=%d", (int)res);
91 osMutexRelease(sd_mutex_id);
92 return false;
93 }
94
95 res = f_mkdir(LOG_DIR);
96 if (res != FR_OK && res != FR_EXIST) {
97 LOG_ERROR("sd_mount: f_mkdir res=%d", (int)res);
98 f_mount(NULL, SDPath, 0);
99 osMutexRelease(sd_mutex_id);
100 return false;
101 }
102
103 if (!find_next_filename()) {
104 f_mount(NULL, SDPath, 0);
105 osMutexRelease(sd_mutex_id);
106 return false;
107 }
108
109 res = f_open(&log_file, filename, FA_WRITE | FA_CREATE_NEW);
110 if (res != FR_OK) {
111 LOG_ERROR("sd_mount: f_open res=%d", (int)res);
112 f_mount(NULL, SDPath, 0);
113 osMutexRelease(sd_mutex_id);
114 return false;
115 }
116
117 if (f_size(&log_file) == 0) {
118 char header[128];
119 int len = board_data_get_header(header, sizeof(header));
120 if (len > 0) {
121 UINT bw;
122 f_write(&log_file, header, len, &bw);
123 f_sync(&log_file);
124 }
125 }
126
127 is_mounted = true;
128 HAL_GPIO_WritePin(SD_STATUS_GPIO_Port, SD_STATUS_Pin, GPIO_PIN_SET);
129
130 osMutexRelease(sd_mutex_id);
131 return true;
132}

164HAL_StatusTypeDef sd_logger_init(void) {
165 log_pool_id = osMemoryPoolNew(SD_QUEUE_LENGTH, sizeof(BoardData_t), &pool_attr);
166 if (log_pool_id == NULL) goto error_exit;
167
168 log_queue_id = osMessageQueueNew(SD_QUEUE_LENGTH, sizeof(BoardData_t*), NULL);
169 if (log_queue_id == NULL) goto error_exit;
170
171 if (FATFS_LinkDriver(&USER_Driver, SDPath) != 0) goto error_exit;
172
173 buffer_free_sem_id[0] = osSemaphoreNew(1, 1, NULL);
174 buffer_free_sem_id[1] = osSemaphoreNew(1, 1, NULL);
175 if (buffer_free_sem_id[0] == NULL || buffer_free_sem_id[1] == NULL) goto error_exit;
176
177 sd_mutex_id = osMutexNew(NULL);
178 if (sd_mutex_id == NULL) goto error_exit;
179
180 packer_task_id = osThreadNew(packer_task_thread, NULL, &packer_attr);
181 if (packer_task_id == NULL) goto error_exit;
182
183 sd_task_id = osThreadNew(sd_task_thread, NULL, &sd_write_attr);
184 if (sd_task_id == NULL) goto error_exit;
185
186#ifdef SD_DETECT_PIN_OPERATIONAL
187 monitor_task_id = osThreadNew(monitor_task_thread, NULL, &monitor_attr);
188 if (monitor_task_id == NULL) goto error_exit;
189#endif
190
191 return HAL_OK;
192
193error_exit:
194 if (packer_task_id) osThreadTerminate(packer_task_id);
195 if (sd_task_id) osThreadTerminate(sd_task_id);
196#ifdef SD_DETECT_PIN_OPERATIONAL
197 if (monitor_task_id) osThreadTerminate(monitor_task_id);
198#endif
199
200 if (sd_mutex_id) osMutexDelete(sd_mutex_id);
201 if (buffer_free_sem_id[0]) osSemaphoreDelete(buffer_free_sem_id[0]);
202 if (buffer_free_sem_id[1]) osSemaphoreDelete(buffer_free_sem_id[1]);
203
204 FATFS_UnLinkDriver(SDPath);
205
206 if (log_queue_id) osMessageQueueDelete(log_queue_id);
207 if (log_pool_id) osMemoryPoolDelete(log_pool_id);
208
209 log_pool_id = NULL;
210 log_queue_id = NULL;
211 sd_mutex_id = NULL;
212 packer_task_id = NULL;
213
214 return HAL_ERROR;
215}

217HAL_StatusTypeDef sd_logger_log_data(const BoardData_t *data) {
218 if (!is_mounted || data == NULL) return HAL_ERROR;

233static void flush_active_buffer(void) {
234 if (active_buffer_pos == 0) return;
235
236 uint8_t ready_idx = active_idx;
237 bytes_to_write[ready_idx] = active_buffer_pos;
238
239 active_idx = !active_idx;
240 active_buffer_pos = 0;
241
242 osSemaphoreAcquire(buffer_free_sem_id[active_idx], osWaitForever);
243 osThreadFlagsSet(sd_task_id, (ready_idx == 0) ? 0x01 : 0x02);
244}

246static void packer_task_thread(void *arg) {
247 (void)arg;
248 BoardData_t *p_item;
249
250 uint32_t last_flush_tick = osKernelGetTickCount();
251 const uint32_t flush_interval = (SD_FORCE_FLUSH_INTERVAL_MS * osKernelGetTickFreq()) / 1000U;
252
253 osSemaphoreAcquire(buffer_free_sem_id[active_idx], osWaitForever);
254 for(;;) {
255 osStatus_t status = osMessageQueueGet(log_queue_id, &p_item, NULL, flush_interval);
256
257 if (status == osOK) {
258 if (active_buffer_pos > (SD_BUFFER_BYTES - SD_BUFFER_MARGIN)) {
259 flush_active_buffer();
260 last_flush_tick = osKernelGetTickCount();
261 }
262
263 int len = board_data_serialize(p_item, &double_buffer[active_idx][active_buffer_pos], SD_BUFFER_BYTES - active_buffer_pos);
264 if (len > 0) active_buffer_pos += (size_t)len;
265 osMemoryPoolFree(log_pool_id, p_item);
266
267 if ((osKernelGetTickCount() - last_flush_tick) >= flush_interval) {
268 flush_active_buffer();
269 last_flush_tick = osKernelGetTickCount();
270 }
271 } else if (status == osErrorTimeout) {
272 flush_active_buffer();
273 last_flush_tick = osKernelGetTickCount();
274 }
275 }

278static void sd_task_thread(void *arg) {
279 (void)arg;
280 uint32_t flags;
281 UINT bw;
282
283#ifndef SD_DETECT_PIN_OPERATIONAL
284 while (!sd_mount()) osDelay(1000);
285#endif
286
287 for(;;) {
288 flags = osThreadFlagsWait(0x03, osFlagsWaitAny, osWaitForever);
289 uint8_t idx = (flags & 0x01) ? 0 : 1;
290
291 osMutexAcquire(sd_mutex_id, osWaitForever);
292 if (is_mounted) {
293 FRESULT wr = f_write(&log_file, double_buffer[idx], bytes_to_write[idx], &bw);
294 if (wr == FR_OK) wr = f_sync(&log_file);
295 if (wr != FR_OK) {
296 LOG_ERROR("sd write fail res=%d, unmounting", (int)wr);
297 f_close(&log_file);
298 f_mount(NULL, SDPath, 0);
299 USER_diskio_invalidate();
300 is_mounted = false;
301 HAL_GPIO_WritePin(SD_STATUS_GPIO_Port, SD_STATUS_Pin, GPIO_PIN_RESET);
302 }
303 }
304 osMutexRelease(sd_mutex_id);
305
306 osSemaphoreRelease(buffer_free_sem_id[idx]);
307 }
308}

324static void sd_detect_sync(bool announce) {
325 bool inserted = (HAL_GPIO_ReadPin(SD_DETECT_GPIO_Port, SD_DETECT_Pin) == GPIO_PIN_RESET);
326 if (inserted && !is_mounted) {
327 if (sd_mount() && announce) sound_play(SOUND_SD_MOUNT);
