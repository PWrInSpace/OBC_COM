249EE_Status FI_DeleteCorruptedFlashAddress(uint32_t Address)
250{
251#ifdef EDATA_ENABLED
252 uint16_t HalfWord[4] = {0U, 0U, 0U, 0U};
253#endif
254 EE_Status status = EE_OK;
255 /* Wait for previous programmation completion */
256 while(__HAL_FLASH_GET_FLAG(FLASH_FLAG_BSY))
257 {
258 }
259
260 /* Clear previous non-secure error programming flag */
261 __HAL_FLASH_CLEAR_FLAG(FLASH_FLAG_ALL_ERRORS);
262
263 /* Set FLASH Programmation bit */
264 SET_BIT(FLASH->NSCR, FLASH_CR_PG);
265
266 /* Program quad word of value 0 */
267
268#ifdef EDATA_ENABLED
269 HAL_FLASH_Program(FLASH_TYPEPROGRAM_HALFWORD_EDATA, Address, (uint32_t)&(HalfWord[0]));
270 HAL_FLASH_Program(FLASH_TYPEPROGRAM_HALFWORD_EDATA, Address+2, (uint32_t)&(HalfWord[1]));
271 HAL_FLASH_Program(FLASH_TYPEPROGRAM_HALFWORD_EDATA, Address+4, (uint32_t)&(HalfWord[2]));
272 HAL_FLASH_Program(FLASH_TYPEPROGRAM_HALFWORD_EDATA, Address+6, (uint32_t)&(HalfWord[3]));
273#else
274 // HAL_FLASH_Program(FLASH_TYPEPROGRAM_QUADWORD, Address, ((uint32_t)QuadWord));
275 *(__IO uint64_t*)(Address) = (uint64_t)0U;
276 *(__IO uint64_t*)(Address+8U) = (uint64_t)0U;
277#endif
278 /* Wait programmation completion */
279 while(__HAL_FLASH_GET_FLAG(FLASH_FLAG_BSY))
280 {
281 }
282
283 /* Check if error occured */
284 if((__HAL_FLASH_GET_FLAG(FLASH_FLAG_WRPERR)) || (__HAL_FLASH_GET_FLAG(FLASH_FLAG_PGSERR)) ||
285 (__HAL_FLASH_GET_FLAG(FLASH_FLAG_STRBERR)) || (__HAL_FLASH_GET_FLAG(FLASH_FLAG_INCERR)))
286 {
287 status = EE_DELETE_ERROR;
288 }
289
290 /* Check FLASH End of Operation flag */
291 if (__HAL_FLASH_GET_FLAG(FLASH_FLAG_EOP))
292 {
293 /* Clear FLASH End of Operation pending bit */
294 __HAL_FLASH_CLEAR_FLAG(FLASH_FLAG_EOP);
295 }
296
297 /* Clear FLASH Programmation bit */
298 CLEAR_BIT(FLASH->NSCR, FLASH_CR_PG);
299
300 /* Clear FLASH ECCD bit */
301 __HAL_FLASH_CLEAR_FLAG(FLASH_FLAG_ECCD);
302
303 return status;
304}

351void OB_Init( void )
352{
353 uint32_t bank = FLASH_BANK_2;
354 uint32_t num_of_edata_page = 1;
355 uint32_t register_value;
356 uint32_t edata_enable = 0;
357
358
359 num_of_edata_page = ((FLASH_EDATA_BASE + (FLASH_EDATA_SIZE>>1)) - START_PAGE_ADDRESS) / FLASH_PAGE_SIZE;
360 register_value = FLASH->EDATA2R_CUR;
361 edata_enable = FLASH_EDATAR_EDATA_EN;
362
363 /* If current EDATA configration is different from new configration, Reprogram Option Byte EDATA settings */
364 if(register_value != (edata_enable | (num_of_edata_page - 1)))
365 {
366 /* Variable used for OB Program procedure */
367 FLASH_OBProgramInitTypeDef FLASH_OBInitStruct;
368
369 /* Current EDATA configuration doesn't match with new configuration. */
370 /* Unlock the Flash option bytes to enable the flash option control register access */
371 HAL_FLASH_OB_Unlock();
372
373 /* Configure 8 sectors for FLASH high-cycle data */
374 FLASH_OBInitStruct.Banks = bank;
375 FLASH_OBInitStruct.OptionType = OPTIONBYTE_EDATA;
376 FLASH_OBInitStruct.EDATASize = edata_enable | num_of_edata_page;
377
378 if(HAL_FLASHEx_OBProgram(&FLASH_OBInitStruct) != HAL_OK)
379 {
380 Error_Handler();
381 }
382
383 /* Start option byte load operation after successful programming operation */
384 HAL_FLASH_OB_Launch();
385
386 /* Lock the Flash control option to restrict register access */
387 HAL_FLASH_OB_Lock();
388 }
389
390}
