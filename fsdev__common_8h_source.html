56// If sharing with CAN, one can set this to be non-zero to give CAN space where it wants it
57// Both of these MUST be a multiple of 2, and are in byte units.
58#ifndef FSDEV_BTABLE_BASE
59 #define FSDEV_BTABLE_BASE 0U
60#endif
61
62TU_VERIFY_STATIC(FSDEV_BTABLE_BASE % 8 == 0, "BTABLE base must be aligned to 8 bytes");
63
64// CFG_TUSB_FSDEV_PMA_SIZE is PMA buffer size in bytes.
65// - 512-byte devices, access with a stride of two words (use every other 16-bit address)
66// - 1024-byte devices, access with a stride of one word (use every 16-bit address) or 32-bit address
67// - 2048-byte devices, access with 32-bit address
68#if CFG_TUSB_FSDEV_PMA_SIZE == 2048 || TU_CHECK_MCU(OPT_MCU_STM32U0)
69 // 32 bit access scheme
70 #define FSDEV_BUS_32BIT
71 #define FSDEV_PMA_STRIDE 1
72 #define pma_access_scheme
73#elif CFG_TUSB_FSDEV_PMA_SIZE == 1024
74 // 2x16 bit / word access scheme
75 #define FSDEV_PMA_STRIDE 1
76 #define pma_access_scheme
77#elif CFG_TUSB_FSDEV_PMA_SIZE == 512
78 // 1x16 bit / word access scheme
79 #define FSDEV_PMA_STRIDE 2
80 #define pma_access_scheme TU_ATTR_ALIGNED(4)
81#endif

105// Buffer Table is located in Packet Memory Area (PMA) and therefore its address access is forced to either
106// 16-bit or 32-bit depending on FSDEV_BUS_32BIT.
107// 0: TX (IN), 1: RX (OUT)
108typedef union {
109 // data is strictly 16-bit access (address could be 32-bit aligned)
110 struct {
111 volatile pma_access_scheme uint16_t addr;
112 volatile pma_access_scheme uint16_t count;
113 } ep16[FSDEV_EP_COUNT][2];
114
115 // strictly 32-bit access
116 struct {
117 volatile uint32_t count_addr;
118 } ep32[FSDEV_EP_COUNT][2];
119} fsdev_btable_t;
120
121TU_VERIFY_STATIC(sizeof(fsdev_btable_t) == FSDEV_EP_COUNT * 8 * FSDEV_PMA_STRIDE, "size is not correct");
122TU_VERIFY_STATIC(FSDEV_BTABLE_BASE + FSDEV_EP_COUNT * 8 <= CFG_TUSB_FSDEV_PMA_SIZE, "BTABLE does not fit in PMA RAM");
123
124#define FSDEV_BTABLE ((volatile fsdev_btable_t *)(FSDEV_PMA_BASE + FSDEV_PMA_STRIDE * (FSDEV_BTABLE_BASE)))
125

139typedef struct {
140 struct {
141 _va32 fsdev_bus_t reg;
142 } ep[FSDEV_EP_COUNT];
143
144 _va32 uint32_t RESERVED7[8]; // Reserved
145 _va32 fsdev_bus_t CNTR; // 40: Control register
146 _va32 fsdev_bus_t ISTR; // 44: Interrupt status register
147 _va32 fsdev_bus_t FNR; // 48: Frame number register
148 _va32 fsdev_bus_t DADDR; // 4C: Device address register
149 _va32 fsdev_bus_t BTABLE; // 50: Buffer Table address register (16-bit only)
150 _va32 fsdev_bus_t LPMCSR; // 54: LPM Control and Status Register (32-bit only)
151 _va32 fsdev_bus_t BCDR; // 58: Battery Charging Detector Register (32-bit only)
152} fsdev_regs_t;
153
154TU_VERIFY_STATIC(offsetof(fsdev_regs_t, CNTR) == 0x40, "Wrong offset");
155TU_VERIFY_STATIC(sizeof(fsdev_regs_t) == 0x5C, "Size is not correct");

199TU_ATTR_ALWAYS_INLINE static inline uint32_t ep_read(uint32_t ep_id) {
200 return FSDEV_REG->ep[ep_id].reg;
201}
202
203TU_ATTR_ALWAYS_INLINE static inline void ep_write(uint32_t ep_id, uint32_t value, bool need_exclusive) {
204 if (need_exclusive) {
205 fsdev_int_disable(0);
206 }
207
208 FSDEV_REG->ep[ep_id].reg = (fsdev_bus_t)value;
209
210 if (need_exclusive) {
211 fsdev_int_enable(0);
212 }
213}

215TU_ATTR_ALWAYS_INLINE static inline void ep_write_clear_ctr(uint32_t ep_id, tusb_dir_t dir) {
216 uint32_t reg = FSDEV_REG->ep[ep_id].reg;
217 reg |= USB_EP_CTR_TX | USB_EP_CTR_RX;
218 reg &= USB_EPREG_MASK;
219 reg &= ~(1 << (USB_EP_CTR_TX_Pos + (dir == TUSB_DIR_IN ? 0 : 8)));
220 ep_write(ep_id, reg, false);
221}

268TU_ATTR_ALWAYS_INLINE static inline uint32_t btable_get_addr(uint32_t ep_id, uint8_t buf_id) {
269#ifdef FSDEV_BUS_32BIT
270 return FSDEV_BTABLE->ep32[ep_id][buf_id].count_addr & 0x0000FFFFu;
271#else
272 return FSDEV_BTABLE->ep16[ep_id][buf_id].addr;
273#endif
274}
275
276TU_ATTR_ALWAYS_INLINE static inline void btable_set_addr(uint32_t ep_id, uint8_t buf_id, uint16_t addr) {
277#ifdef FSDEV_BUS_32BIT
278 uint32_t count_addr = FSDEV_BTABLE->ep32[ep_id][buf_id].count_addr;
279 count_addr = (count_addr & 0xFFFF0000u) | (addr & 0x0000FFFCu);
280 FSDEV_BTABLE->ep32[ep_id][buf_id].count_addr = count_addr;
281#else
282 FSDEV_BTABLE->ep16[ep_id][buf_id].addr = addr;
283#endif
284}
285
286TU_ATTR_ALWAYS_INLINE static inline uint16_t btable_get_count(uint32_t ep_id, uint8_t buf_id) {
287 uint16_t count;
288#ifdef FSDEV_BUS_32BIT
289 count = (FSDEV_BTABLE->ep32[ep_id][buf_id].count_addr >> 16);
290#else
291 count = FSDEV_BTABLE->ep16[ep_id][buf_id].count;
292#endif
293 return count & 0x3FFU;
294}
295
296TU_ATTR_ALWAYS_INLINE static inline void btable_set_count(uint32_t ep_id, uint8_t buf_id, uint16_t byte_count) {
297#ifdef FSDEV_BUS_32BIT
298 uint32_t count_addr = FSDEV_BTABLE->ep32[ep_id][buf_id].count_addr;
299 count_addr = (count_addr & ~0x03FF0000u) | ((byte_count & 0x3FFu) << 16);
300 FSDEV_BTABLE->ep32[ep_id][buf_id].count_addr = count_addr;
301#else
302 uint16_t cnt = FSDEV_BTABLE->ep16[ep_id][buf_id].count;
303 cnt = (cnt & ~0x3FFU) | (byte_count & 0x3FFU);
304 FSDEV_BTABLE->ep16[ep_id][buf_id].count = cnt;
305#endif
306}
