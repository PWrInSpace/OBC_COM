187// MCU specific PHY init, called BEFORE core reset
188// - dwc2 3.30a (H5) use USB_HS_PHYC
189// - dwc2 4.11a (U5) use femtoPHY
190// - dwc2 x.xxx (WBA) use USB_OTG_HS
191static inline void dwc2_phy_init(dwc2_regs_t* dwc2, uint8_t hs_phy_type) {
192 if (hs_phy_type == GHWCFG2_HSPHY_NOT_SUPPORTED) {
193 // Enable on-chip FS PHY
194 dwc2->stm32_gccfg |= STM32_GCCFG_PWRDWN;
195
196 // https://community.st.com/t5/stm32cubemx-mcus/why-stm32h743-usb-fs-doesn-t-work-if-freertos-tickless-idle/m-p/349480#M18867
197 // H7 running on full-speed phy need to disable ULPI clock in sleep mode.
198 // Otherwise, USB won't work when mcu executing WFI/WFE instruction i.e tick-less RTOS.
199 // Note: there may be other family that is affected by this, but only H7 and F7 is tested so far
200 #if defined(USB_OTG_FS_PERIPH_BASE) && defined(RCC_AHB1LPENR_USB2OTGFSULPILPEN)
201 if ( USB_OTG_FS_PERIPH_BASE == (uint32_t) dwc2 ) {
202 RCC->AHB1LPENR &= ~RCC_AHB1LPENR_USB2OTGFSULPILPEN;
203 }
204 #endif
205
206 #if defined(USB_OTG_HS_PERIPH_BASE) && defined(RCC_AHB1LPENR_USB1OTGHSULPILPEN)
207 if ( USB_OTG_HS_PERIPH_BASE == (uint32_t) dwc2 ) {
208 RCC->AHB1LPENR &= ~RCC_AHB1LPENR_USB1OTGHSULPILPEN;
209 }
210 #endif
211
212 #if defined(USB_OTG_HS_PERIPH_BASE) && defined(RCC_AHB1LPENR_OTGHSULPILPEN)
213 if ( USB_OTG_HS_PERIPH_BASE == (uint32_t) dwc2 ) {
214 RCC->AHB1LPENR &= ~RCC_AHB1LPENR_OTGHSULPILPEN;
215 }
216 #endif
217
218 } else {
219#if CFG_TUSB_MCU != OPT_MCU_STM32U5 && CFG_TUSB_MCU != OPT_MCU_STM32WBA
220 // Disable FS PHY, TODO on U5A5 (dwc2 4.11a) 16th bit is 'Host CDP behavior enable'
221 dwc2->stm32_gccfg &= ~STM32_GCCFG_PWRDWN;
222#endif
223 // Enable on-chip HS PHY
224 if (hs_phy_type == GHWCFG2_HSPHY_UTMI || hs_phy_type == GHWCFG2_HSPHY_UTMI_ULPI) {
225 #ifdef USB_HS_PHYC
226 // Enable UTMI HS PHY
227 dwc2->stm32_gccfg |= STM32_GCCFG_PHYHSEN;
228
229 // Enable LDO
230 USB_HS_PHYC->USB_HS_PHYC_LDO |= USB_HS_PHYC_LDO_ENABLE;
231
232 // Wait until LDO ready
233 while ( 0 == (USB_HS_PHYC->USB_HS_PHYC_LDO & USB_HS_PHYC_LDO_STATUS) ) {}
234
235 uint32_t phyc_pll = 0;
236
237 // TODO Try to get HSE_VALUE from registers instead of depending CFLAGS
238 switch ( HSE_VALUE )
239 {
240 case 12000000: phyc_pll = USB_HS_PHYC_PLL1_PLLSEL_12MHZ ; break;
241 case 12500000: phyc_pll = USB_HS_PHYC_PLL1_PLLSEL_12_5MHZ ; break;
242 case 16000000: phyc_pll = USB_HS_PHYC_PLL1_PLLSEL_16MHZ ; break;
243 case 24000000: phyc_pll = USB_HS_PHYC_PLL1_PLLSEL_24MHZ ; break;
244 case 25000000: phyc_pll = USB_HS_PHYC_PLL1_PLLSEL_25MHZ ; break;
245 case 32000000: phyc_pll = USB_HS_PHYC_PLL1_PLLSEL_Msk ; break; // Value not defined in header
246 default:
247 TU_ASSERT(false, );
248 }
249 USB_HS_PHYC->USB_HS_PHYC_PLL = phyc_pll;
250
251 // Control the tuning interface of the High Speed PHY
252 // Use magic value (USB_HS_PHYC_TUNE_VALUE) from ST driver for F7
253 USB_HS_PHYC->USB_HS_PHYC_TUNE |= 0x00000F13U;
254
255 // Enable PLL internal PHY
256 USB_HS_PHYC->USB_HS_PHYC_PLL |= USB_HS_PHYC_PLL_PLLEN;
257
258 // Wait ~2ms until the PLL is ready (there's no RDY bit to query)
259 tusb_time_delay_ms_api(2);
260 #else
261
262 #endif
263 }
264 }

267// MCU specific PHY deinit, disable PHY power
268static inline void dwc2_phy_deinit(dwc2_regs_t* dwc2, uint8_t hs_phy_type) {
269 if (hs_phy_type == GHWCFG2_HSPHY_NOT_SUPPORTED) {
270 // Disable on-chip FS PHY
271 dwc2->stm32_gccfg &= ~STM32_GCCFG_PWRDWN;
272 } else {
273 // Disable HS PHY
274 #ifdef USB_HS_PHYC
275 dwc2->stm32_gccfg &= ~STM32_GCCFG_PHYHSEN;
276 // Disable PLL and LDO
277 USB_HS_PHYC->USB_HS_PHYC_PLL &= ~USB_HS_PHYC_PLL_PLLEN;
278 USB_HS_PHYC->USB_HS_PHYC_LDO &= ~USB_HS_PHYC_LDO_ENABLE;
279 #endif
280 }
281}

283// MCU specific PHY update, it is called AFTER init() and core reset
284static inline void dwc2_phy_update(dwc2_regs_t* dwc2, uint8_t hs_phy_type) {
285 // used to set turnaround time for fullspeed, nothing to do in highspeed mode
286 if (hs_phy_type == GHWCFG2_HSPHY_NOT_SUPPORTED) {
287 // Turnaround timeout depends on the AHB clock dictated by STM32 Reference Manual
288 uint32_t turnaround;
289
290 if (SystemCoreClock >= 32000000u) {
291 turnaround = 0x6u;
292 } else if (SystemCoreClock >= 27500000u) {
293 turnaround = 0x7u;
294 } else if (SystemCoreClock >= 24000000u) {
295 turnaround = 0x8u;
296 } else if (SystemCoreClock >= 21800000u) {
297 turnaround = 0x9u;
298 }
299 else if (SystemCoreClock >= 20000000u) {
300 turnaround = 0xAu;
301 }
302 else if (SystemCoreClock >= 18500000u) {
303 turnaround = 0xBu;
304 }
305 else if (SystemCoreClock >= 17200000u) {
306 turnaround = 0xCu;
307 }
308 else if (SystemCoreClock >= 16000000u) {
309 turnaround = 0xDu;
310 }
311 else if (SystemCoreClock >= 15000000u) {
312 turnaround = 0xEu;
313 }
314 else {
315 turnaround = 0xFu;
316 }
317
318 dwc2->gusbcfg = (dwc2->gusbcfg & ~GUSBCFG_TRDT_Msk) | (turnaround << GUSBCFG_TRDT_Pos);
319 }
320}

322//------------- GCCFG configuration -------------//
323static inline void dwc2_stm32_gccfg_cfg(dwc2_regs_t* dwc2, bool vbus_sensing, bool is_host) {
324 if (is_host) {
325 vbus_sensing = false;
326 }
327
328 uint32_t gccfg = dwc2->stm32_gccfg;
329 if (dwc2->guid < 0x2000) {
330 // use VBUSASEN/VBUSBSEN/NOVBUSSENS bits
331 if (is_host) {
332 gccfg &= ~(STM32_GCCFG_NOVBUSSENS | STM32_GCCFG_VBUSBSEN | STM32_GCCFG_VBUSASEN);
333 } else {
334 if (vbus_sensing) {
335 gccfg &= ~STM32_GCCFG_NOVBUSSENS;
336 gccfg |= STM32_GCCFG_VBUSBSEN;
337 } else {
338 gccfg |= STM32_GCCFG_NOVBUSSENS;
339 gccfg &= ~(STM32_GCCFG_VBUSBSEN | STM32_GCCFG_VBUSASEN);
340 }
341 }
342 } else if (dwc2->guid < 0x5000) {
343 // the later version uses VBDEN with battery charging detection
344 if (vbus_sensing) {
345 gccfg |= STM32_GCCFG_VBDEN;
346 } else {
347 gccfg &= ~STM32_GCCFG_VBDEN;
348 }
349 } else {
350 // from 0x5000 ST seems to use femtoPHY for UTMI+ HS PHY. Which use VBVALEXTOEN and VBVALOVAL for software override
351 // external VBUS sensing
352 // Note: N6 does not support hardware VBUS sensing, so the software override is always active. Therefore, VBDEN and
353 // VBVALEXTOEN are not available
354#if CFG_TUSB_MCU == OPT_MCU_STM32N6
355 if (is_host) {
356 gccfg |= STM32_GCCFG_PULLDOWNEN;
357 gccfg &= ~(STM32_GCCFG_VBVALOVAL);
358 } else {
359 gccfg &= ~STM32_GCCFG_PULLDOWNEN;
360 gccfg |= STM32_GCCFG_VBVALOVAL;
361 }
362#else
363 if (is_host) {
364 gccfg |= STM32_GCCFG_PULLDOWNEN;
365 gccfg &= ~(STM32_GCCFG_VBDEN | STM32_GCCFG_VBVALEXTOEN | STM32_GCCFG_VBVALOVAL);
366 } else {
367 gccfg &= ~STM32_GCCFG_PULLDOWNEN;
368 if (vbus_sensing) {
369 gccfg |= STM32_GCCFG_VBDEN;
370 gccfg &= ~(STM32_GCCFG_VBVALEXTOEN | STM32_GCCFG_VBVALOVAL);
371 } else {
372 gccfg &= ~STM32_GCCFG_VBDEN;
373 gccfg |= STM32_GCCFG_VBVALEXTOEN | STM32_GCCFG_VBVALOVAL;
374 }
375 }
376#endif
377 }
378
379 dwc2->stm32_gccfg = gccfg;
380}

416TU_ATTR_ALWAYS_INLINE static inline uint32_t round_up_to_cache_line_size(uint32_t size) {
417 if (size & (CFG_TUSB_MEM_DCACHE_LINE_SIZE_DEFAULT-1)) {
418 size = (size & ~(CFG_TUSB_MEM_DCACHE_LINE_SIZE_DEFAULT-1)) + CFG_TUSB_MEM_DCACHE_LINE_SIZE_DEFAULT;
419 }
420 return size;
421}

423TU_ATTR_ALWAYS_INLINE static inline bool is_cache_mem(uintptr_t addr) {
424 if (0 == (SCB->CCR & SCB_CCR_DC_Msk)) {
425 return false; // D-Cache is disabled
426 }
427 for (unsigned int i = 0; i < TU_ARRAY_SIZE(uncached_regions); i++) {
428 if (uncached_regions[i].start <= addr && addr <= uncached_regions[i].end) { return false; }
429 }
430 return true;
431}

436 data_size = round_up_to_cache_line_size(data_size);
