75// Get local (cached) line state
76// This function should return correct values if tuh_cdc_set_control_line_state() / tuh_cdc_get_control_line_state()
77// are invoked previously or CFG_TUH_CDC_LINE_STATE_ON_ENUM is defined.
78bool tuh_cdc_get_control_line_state_local(uint8_t idx, uint16_t *line_state);

80// Get current DTR status
81TU_ATTR_ALWAYS_INLINE static inline bool tuh_cdc_get_dtr(uint8_t idx) {
82 uint16_t line_state;
83 TU_VERIFY(tuh_cdc_get_control_line_state_local(idx, &line_state));
84 return (line_state & CDC_CONTROL_LINE_STATE_DTR) != 0;
85}
86
87// Get current RTS status
88TU_ATTR_ALWAYS_INLINE static inline bool tuh_cdc_get_rts(uint8_t idx) {
89 uint16_t line_state;
90 TU_VERIFY(tuh_cdc_get_control_line_state_local(idx, &line_state));
91 return (line_state & CDC_CONTROL_LINE_STATE_RTS) != 0;
92}
93
94// Check if interface is connected (DTR active)
95TU_ATTR_ALWAYS_INLINE static inline bool tuh_cdc_connected(uint8_t idx) {
96 return tuh_cdc_get_dtr(idx);
97}

139//--------------------------------------------------------------------+
140// Control Request API
141// Each Function will make a USB control transfer request to/from device
142// - If complete_cb is provided, the function will return immediately and invoke
143// the callback when request is complete.
144// - If complete_cb is NULL, the function will block until request is complete.
145// In this case, user_data should be usb_xfer_result_t* to hold the transfer result.

148// Request to Set Control Line State: DTR (bit 0), RTS (bit 1)
149bool tuh_cdc_set_control_line_state(uint8_t idx, uint16_t line_state, tuh_xfer_cb_t complete_cb, uintptr_t user_data);

151// Request to Set DTR
152TU_ATTR_ALWAYS_INLINE static inline bool tuh_cdc_set_dtr(uint8_t idx, bool dtr_state, tuh_xfer_cb_t complete_cb,
153 uintptr_t user_data) {
154 cdc_line_control_state_t line_state = {.dtr = dtr_state};
155 line_state.rts = tuh_cdc_get_rts(idx);
156 return tuh_cdc_set_control_line_state(idx, line_state.value, complete_cb, user_data);
157}
158
159// Request to Set RTS
160TU_ATTR_ALWAYS_INLINE static inline bool tuh_cdc_set_rts(uint8_t idx, bool rts_state, tuh_xfer_cb_t complete_cb,
161 uintptr_t user_data) {
162 cdc_line_control_state_t line_state = {.rts = rts_state};
163 line_state.dtr = tuh_cdc_get_dtr(idx);
164 return tuh_cdc_set_control_line_state(idx, line_state.value, complete_cb, user_data);
165}

180// Should only use if tuh_cdc_set_line_coding() / tuh_cdc_get_line_coding() never got invoked and

184// Connect by set both DTR, RTS
185TU_ATTR_ALWAYS_INLINE static inline bool tuh_cdc_connect(uint8_t idx, tuh_xfer_cb_t complete_cb, uintptr_t user_data) {
186 return tuh_cdc_set_control_line_state(idx, CDC_CONTROL_LINE_STATE_DTR | CDC_CONTROL_LINE_STATE_RTS, complete_cb,
187 user_data);
188}
189
190// Disconnect by clear both DTR, RTS
191TU_ATTR_ALWAYS_INLINE static inline bool tuh_cdc_disconnect(uint8_t idx, tuh_xfer_cb_t complete_cb,
192 uintptr_t user_data) {
193 return tuh_cdc_set_control_line_state(idx, 0x00, complete_cb, user_data);
194}
