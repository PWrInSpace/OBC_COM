141// data in response of HUB_REQUEST_GET_STATUS, wIndex = Port num
142typedef struct {
143 union TU_ATTR_PACKED {
144 struct TU_ATTR_PACKED {
145 // Bit 0-4 are for change & status
146 uint16_t connection : 1; // [0] 0 = no device, 1 = device connected
147 uint16_t port_enable : 1; // [1] port is enabled
148 uint16_t suspend : 1; // [2]
149 uint16_t over_current : 1; // [3] over-current exists
150 uint16_t reset : 1; // [4] 0 = no reset, 1 = resetting
151
152 // From Bit 5 are for status only
153 uint16_t rsv5_7 : 3; // [5..7] reserved
154 uint16_t port_power : 1; // [8] 0 = port is off, 1 = port is on
155 uint16_t low_speed : 1; // [9] low speed device attached
156 uint16_t high_speed : 1; // [10] high speed device attached
157 uint16_t port_test_mode : 1; // [11] port in test mode
158 uint16_t port_indicator_control : 1; // [12] 0: default color, 1: indicator is software controlled
159 uint16_t TU_RESERVED : 3; // [13..15] reserved
160 };
161
162 uint16_t value;
163 } status, change;
164} hub_port_status_response_t;
165TU_VERIFY_STATIC( sizeof(hub_port_status_response_t) == 4, "size is not correct");

190// Reset a port
191TU_ATTR_ALWAYS_INLINE static inline
192bool hub_port_reset(uint8_t hub_addr, uint8_t hub_port, tuh_xfer_cb_t complete_cb, uintptr_t user_data) {
193 return hub_port_set_feature(hub_addr, hub_port, HUB_FEATURE_PORT_RESET, complete_cb, user_data);
194}

196// Clear Port Reset Change
197TU_ATTR_ALWAYS_INLINE static inline
198bool hub_port_clear_reset_change(uint8_t hub_addr, uint8_t hub_port, tuh_xfer_cb_t complete_cb, uintptr_t user_data) {
199 return hub_port_clear_feature(hub_addr, hub_port, HUB_FEATURE_PORT_RESET_CHANGE, complete_cb, user_data);
200}

202// Get Hub status (port = 0)
203TU_ATTR_ALWAYS_INLINE static inline
204bool hub_get_status(uint8_t hub_addr, void* resp, tuh_xfer_cb_t complete_cb, uintptr_t user_data) {
205 return hub_port_get_status(hub_addr, 0, resp, complete_cb, user_data);
206}

211 return hub_port_clear_feature(hub_addr, 0, feature, complete_cb, user_data);
