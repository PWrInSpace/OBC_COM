130static DRESULT SD_ioctl(BYTE lun, BYTE cmd, void *buff) {
131 DRESULT res = RES_ERROR;
132 HAL_SD_CardInfoTypeDef CardInfo;
133
134 if (Stat & STA_NOINIT)
135 return RES_NOTRDY;
136
137 switch (cmd) {
138 /* Make sure that no pending write process */
139 case CTRL_SYNC:
140 res = RES_OK;
141 break;
142
143 /* Get number of sectors on the disk (DWORD) */
144 case GET_SECTOR_COUNT:
145 HAL_SD_GetCardInfo(&sdmmc_handle, &CardInfo);
146 *(DWORD *)buff = CardInfo.LogBlockNbr;
147 res = RES_OK;
148 break;
149
150 /* Get R/W sector size (WORD) */
151 case GET_SECTOR_SIZE:
152 HAL_SD_GetCardInfo(&sdmmc_handle, &CardInfo);
153 *(WORD *)buff = CardInfo.LogBlockSize;
154 res = RES_OK;
155 break;
156
157 /* Get erase block size in unit of sector (DWORD) */
158 case GET_BLOCK_SIZE:
159 HAL_SD_GetCardInfo(&sdmmc_handle, &CardInfo);
160 *(DWORD *)buff = CardInfo.LogBlockSize / BLOCKSIZE;
161 res = RES_OK;
162 break;
163
164 default:
165 res = RES_PARERR;
166 }
167
168 return res;
