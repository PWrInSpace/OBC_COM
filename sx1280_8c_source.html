282void SX1280SetModulationParams( SX1280_t *sx1280, ModulationParams_t *modulationParams )
283{
284 uint8_t buf[3];
285
286 // Check if required configuration corresponds to the stored packet type
287 // If not, silently update radio packet type
288 if( PacketType != modulationParams->PacketType )
289 {
290 SX1280SetPacketType( sx1280, modulationParams->PacketType );
291 }
292
293 switch( modulationParams->PacketType )
294 {
295 case PACKET_TYPE_GFSK:
296 buf[0] = modulationParams->Params.Gfsk.BitrateBandwidth;
297 buf[1] = modulationParams->Params.Gfsk.ModulationIndex;
298 buf[2] = modulationParams->Params.Gfsk.ModulationShaping;
299 break;
300
301 case PACKET_TYPE_LORA:
302 case PACKET_TYPE_RANGING:
303 buf[0] = modulationParams->Params.LoRa.SpreadingFactor;
304 buf[1] = modulationParams->Params.LoRa.Bandwidth;
305 buf[2] = modulationParams->Params.LoRa.CodingRate;
306 LoRaBandwidth = modulationParams->Params.LoRa.Bandwidth;
307 break;
308
309 case PACKET_TYPE_FLRC:
310 buf[0] = modulationParams->Params.Flrc.BitrateBandwidth;
311 buf[1] = modulationParams->Params.Flrc.CodingRate;
312 buf[2] = modulationParams->Params.Flrc.ModulationShaping;
313 break;
314
315 case PACKET_TYPE_BLE:
316 buf[0] = modulationParams->Params.Ble.BitrateBandwidth;
317 buf[1] = modulationParams->Params.Ble.ModulationIndex;
318 buf[2] = modulationParams->Params.Ble.ModulationShaping;
319 break;
320
321 case PACKET_TYPE_NONE:
322 buf[0] = 0;
323 buf[1] = 0;
324 buf[2] = 0;
325 break;
326 }
327 SX1280HalWriteCommand(sx1280, RADIO_SET_MODULATIONPARAMS, buf, 3 );
328}

397void SX1280GetRxBufferStatus( SX1280_t *sx1280, uint8_t *payloadLength, uint8_t *rxStartBufferPointer )
398{
399 uint8_t status[2];
400
401 SX1280HalReadCommand(sx1280, RADIO_GET_RXBUFFERSTATUS, status, 2 );
402
403 // In case of LORA fixed header, the payloadLength is obtained by reading
404 // the register REG_LR_PAYLOADLENGTH
405 if( ( SX1280GetPacketType( sx1280 ) == PACKET_TYPE_LORA ) && ( SX1280HalReadRegister( sx1280, REG_LR_PACKETPARAMS ) >> 7 == 1 ) )
406 {
407 *payloadLength = SX1280HalReadRegister( sx1280, REG_LR_PAYLOADLENGTH );
408 }
409 else if( SX1280GetPacketType( sx1280 ) == PACKET_TYPE_BLE )
410 {
411 // In the case of BLE, the size returned in status[0] do not include the 2-byte length PDU header
412 // so it is added there
413 *payloadLength = status[0] + 2;
414 }
415 else
416 {
417 *payloadLength = status[0];
418 }
419
420 *rxStartBufferPointer = status[1];
421}

423void SX1280GetPacketStatus( SX1280_t *sx1280, PacketStatus_t *pktStatus )
424{
425 uint8_t status[5];
426
427 SX1280HalReadCommand(sx1280, RADIO_GET_PACKETSTATUS, status, 5 );
428
429 pktStatus->packetType = SX1280GetPacketType( sx1280 );
430 switch( pktStatus->packetType )
431 {
432 case PACKET_TYPE_GFSK:
433 pktStatus->Params.Gfsk.RssiAvg = -status[0] / 2;
434 pktStatus->Params.Gfsk.RssiSync = -status[1] / 2;
435
436 pktStatus->Params.Gfsk.ErrorStatus.SyncError = ( status[2] >> 6 ) & 0x01;
437 pktStatus->Params.Gfsk.ErrorStatus.LengthError = ( status[2] >> 5 ) & 0x01;
438 pktStatus->Params.Gfsk.ErrorStatus.CrcError = ( status[2] >> 4 ) & 0x01;
439 pktStatus->Params.Gfsk.ErrorStatus.AbortError = ( status[2] >> 3 ) & 0x01;
440 pktStatus->Params.Gfsk.ErrorStatus.HeaderReceived = ( status[2] >> 2 ) & 0x01;
441 pktStatus->Params.Gfsk.ErrorStatus.PacketReceived = ( status[2] >> 1 ) & 0x01;
442 pktStatus->Params.Gfsk.ErrorStatus.PacketControlerBusy = status[2] & 0x01;
443
444 pktStatus->Params.Gfsk.TxRxStatus.RxNoAck = ( status[3] >> 5 ) & 0x01;
445 pktStatus->Params.Gfsk.TxRxStatus.PacketSent = status[3] & 0x01;
446
447 pktStatus->Params.Gfsk.SyncAddrStatus = status[4] & 0x07;
448 break;
449
450 case PACKET_TYPE_LORA:
451 case PACKET_TYPE_RANGING:
452 pktStatus->Params.LoRa.RssiPkt = -status[0] / 2;
453 ( status[1] < 128 ) ? ( pktStatus->Params.LoRa.SnrPkt = status[1] / 4 ) : ( pktStatus->Params.LoRa.SnrPkt = ( ( status[1] - 256 ) /4 ) );
454
455 pktStatus->Params.LoRa.ErrorStatus.SyncError = ( status[2] >> 6 ) & 0x01;
456 pktStatus->Params.LoRa.ErrorStatus.LengthError = ( status[2] >> 5 ) & 0x01;
457 pktStatus->Params.LoRa.ErrorStatus.CrcError = ( status[2] >> 4 ) & 0x01;
458 pktStatus->Params.LoRa.ErrorStatus.AbortError = ( status[2] >> 3 ) & 0x01;
459 pktStatus->Params.LoRa.ErrorStatus.HeaderReceived = ( status[2] >> 2 ) & 0x01;
460 pktStatus->Params.LoRa.ErrorStatus.PacketReceived = ( status[2] >> 1 ) & 0x01;
461 pktStatus->Params.LoRa.ErrorStatus.PacketControlerBusy = status[2] & 0x01;
462
463 pktStatus->Params.LoRa.TxRxStatus.RxNoAck = ( status[3] >> 5 ) & 0x01;
464 pktStatus->Params.LoRa.TxRxStatus.PacketSent = status[3] & 0x01;
465
466 pktStatus->Params.LoRa.SyncAddrStatus = status[4] & 0x07;
467 break;
468
469 case PACKET_TYPE_FLRC:
470 pktStatus->Params.Flrc.RssiAvg = -status[0] / 2;
471 pktStatus->Params.Flrc.RssiSync = -status[1] / 2;
472
473 pktStatus->Params.Flrc.ErrorStatus.SyncError = ( status[2] >> 6 ) & 0x01;
474 pktStatus->Params.Flrc.ErrorStatus.LengthError = ( status[2] >> 5 ) & 0x01;
475 pktStatus->Params.Flrc.ErrorStatus.CrcError = ( status[2] >> 4 ) & 0x01;
476 pktStatus->Params.Flrc.ErrorStatus.AbortError = ( status[2] >> 3 ) & 0x01;
477 pktStatus->Params.Flrc.ErrorStatus.HeaderReceived = ( status[2] >> 2 ) & 0x01;
478 pktStatus->Params.Flrc.ErrorStatus.PacketReceived = ( status[2] >> 1 ) & 0x01;
479 pktStatus->Params.Flrc.ErrorStatus.PacketControlerBusy = status[2] & 0x01;
480
481 pktStatus->Params.Flrc.TxRxStatus.RxPid = ( status[3] >> 6 ) & 0x03;
482 pktStatus->Params.Flrc.TxRxStatus.RxNoAck = ( status[3] >> 5 ) & 0x01;
483 pktStatus->Params.Flrc.TxRxStatus.RxPidErr = ( status[3] >> 4 ) & 0x01;
484 pktStatus->Params.Flrc.TxRxStatus.PacketSent = status[3] & 0x01;
485
486 pktStatus->Params.Flrc.SyncAddrStatus = status[4] & 0x07;
487 break;
488
489 case PACKET_TYPE_BLE:
490 pktStatus->Params.Ble.RssiAvg = -status[0] / 2;
491 pktStatus->Params.Ble.RssiSync = -status[1] / 2;
492
493 pktStatus->Params.Ble.ErrorStatus.SyncError = ( status[2] >> 6 ) & 0x01;
494 pktStatus->Params.Ble.ErrorStatus.LengthError = ( status[2] >> 5 ) & 0x01;
495 pktStatus->Params.Ble.ErrorStatus.CrcError = ( status[2] >> 4 ) & 0x01;
496 pktStatus->Params.Ble.ErrorStatus.AbortError = ( status[2] >> 3 ) & 0x01;
497 pktStatus->Params.Ble.ErrorStatus.HeaderReceived = ( status[2] >> 2 ) & 0x01;
498 pktStatus->Params.Ble.ErrorStatus.PacketReceived = ( status[2] >> 1 ) & 0x01;
499 pktStatus->Params.Ble.ErrorStatus.PacketControlerBusy = status[2] & 0x01;
500
501 pktStatus->Params.Ble.TxRxStatus.PacketSent = status[3] & 0x01;
502
503 pktStatus->Params.Ble.SyncAddrStatus = status[4] & 0x07;
504 break;
505
506 case PACKET_TYPE_NONE:
507 // In that specific case, we set everything in the pktStatus to zeros
508 // and reset the packet type accordingly
509 memset( pktStatus, 0, sizeof( PacketStatus_t ) );
510 pktStatus->packetType = PACKET_TYPE_NONE;
511 break;
512 }
513}

851double SX1280GetRangingResult( SX1280_t *sx1280, RadioRangingResultTypes_t resultType )
852{
853 uint32_t valLsb = 0;
854 double val = 0.0;
855
856 switch( SX1280GetPacketType( sx1280 ) )
857 {
858 case PACKET_TYPE_RANGING:
859 SX1280SetStandby( sx1280, STDBY_XOSC );
860 SX1280HalWriteRegister( sx1280, 0x97F, SX1280HalReadRegister( sx1280, 0x97F ) | ( 1 << 1 ) ); // enable LORA modem clock
861 SX1280HalWriteRegister( sx1280, REG_LR_RANGINGRESULTCONFIG, ( SX1280HalReadRegister( sx1280, REG_LR_RANGINGRESULTCONFIG ) & MASK_RANGINGMUXSEL ) | ( ( ( ( uint8_t )resultType ) & 0x03 ) << 4 ) );
862 valLsb = ( ( SX1280HalReadRegister( sx1280, REG_LR_RANGINGRESULTBASEADDR ) << 16 ) | ( SX1280HalReadRegister( sx1280, REG_LR_RANGINGRESULTBASEADDR + 1 ) << 8 ) | ( SX1280HalReadRegister( sx1280, REG_LR_RANGINGRESULTBASEADDR + 2 ) ) );
863 SX1280SetStandby( sx1280, STDBY_RC );
864
865 // Convertion from LSB to distance. For explanation on the formula, refer to Datasheet of SX1280
866 switch( resultType )
867 {
868 case RANGING_RESULT_RAW:
869 // Convert the ranging LSB to distance in meter
870 val = ( double )SX1280complement2( valLsb, 24 ) / ( double )SX1280GetLoRaBandwidth( sx1280 ) * 36621.09375;
871 break;
872
873 case RANGING_RESULT_AVERAGED:
874 case RANGING_RESULT_DEBIASED:
875 case RANGING_RESULT_FILTERED:
876 val = ( double )valLsb * 20.0 / 100.0;
877 break;
878
879 default:
880 val = 0.0;
881 }
882 break;
883
884 default:
885 break;
886 }
887 return val;
888}

964int8_t SX1280GetHexFileLineFields( SX1280_t *sx1280, char* line, uint8_t *bytes, uint16_t *addr, uint16_t *num, uint8_t *code )
965{
966 uint16_t sum, len, cksum;
967 char *ptr;
968
969 *num = 0;
970 if( line[0] != ':' )
971 {
972 return 0;
973 }
974 if( strlen( line ) < 11 )
975 {
976 return 0;
977 }
978 ptr = line + 1;
979 if( !sscanf( ptr, "%02hx", &len ) )
980 {
981 return 0;
982 }
983 ptr += 2;
984 if( strlen( line ) < ( 11 + ( len * 2 ) ) )
985 {
986 return 0;
987 }
988 if( !sscanf( ptr, "%04hx", addr ) )
989 {
990 return 0;
991 }
992 ptr += 4;
993 if( !sscanf( ptr, "%02hhx", code ) )
994 {
995 return 0;
996 }
997 ptr += 2;
998 sum = ( len & 255 ) + ( ( *addr >> 8 ) & 255 ) + ( *addr & 255 ) + ( ( *code >> 8 ) & 255 ) + ( *code & 255 );
999 while( *num != len )
1000 {
1001 if( !sscanf( ptr, "%02hhx", &bytes[*num] ) )
1002 {
1003 return 0;
1004 }
1005 ptr += 2;
1006 sum += bytes[*num] & 255;
1007 ( *num )++;
1008 if( *num >= 256 )
1009 {
1010 return 0;
1011 }
1012 }
1013 if( !sscanf( ptr, "%02hx", &cksum ) )
1014 {
1015 return 0;
1016 }
1017 if( ( ( sum & 255 ) + ( cksum & 255 ) ) & 255 )
1018 {
1019 return 0; // checksum error
1020 }
1021
1022 return 1;
1023}

1204void SX1280ProcessIrqs( SX1280_t *sx1280)
1205{
1206 RadioPacketTypes_t packetType = PACKET_TYPE_NONE;
1207
1208 if( SX1280GetOpMode( sx1280 ) == SX1280_MODE_SLEEP )
1209 {
1210 return; // DIO glitch on V2b :-)
1211 }
1212
1213 if( PollingMode == true )
1214 {
1215 if( IrqState == true )
1216 {
1217 DISABLE_IRQ( );
1218 IrqState = false;
1219 ENABLE_IRQ( );
1220 }
1221 else
1222 {
1223 return;
1224 }
1225 }
1226
1227 packetType = SX1280GetPacketType( sx1280 );
1228 uint16_t irqRegs = SX1280GetIrqStatus( sx1280 );
1229 SX1280ClearIrqStatus( sx1280, IRQ_RADIO_ALL );
1230
1231 switch( packetType )
1232 {
1233 case PACKET_TYPE_GFSK:
1234 case PACKET_TYPE_FLRC:
1235 case PACKET_TYPE_BLE:
1236 switch( OperatingMode )
1237 {
1238 case MODE_RX:
1239 if( ( irqRegs & IRQ_RX_DONE ) == IRQ_RX_DONE )
1240 {
