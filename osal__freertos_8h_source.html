86TU_ATTR_ALWAYS_INLINE static inline uint32_t _osal_ms2tick(uint32_t msec) {
87 if (msec == OSAL_TIMEOUT_WAIT_FOREVER) { return portMAX_DELAY; }
88 if (msec == 0) { return 0; }
89
90 uint32_t ticks = pdMS_TO_TICKS(msec);
91
92 // If configTICK_RATE_HZ is less than 1000 and 1 tick > 1 ms, we still need to delay at least 1 tick
93 if (ticks == 0) { ticks = 1; }
94
95 return ticks;
96}

100}

187TU_ATTR_ALWAYS_INLINE static inline bool osal_semaphore_post(osal_semaphore_t sem_hdl, bool in_isr) {
188 if (!in_isr) {
189 return xSemaphoreGive(sem_hdl) != 0;
190 } else {
191 BaseType_t xHigherPriorityTaskWoken = pdFALSE;
192 BaseType_t res = xSemaphoreGiveFromISR(sem_hdl, &xHigherPriorityTaskWoken);
193 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
194 return res != 0;
195 }
196}
